export default function AcadivoAssistant({onClose}:{onClose:()=>void}) {
  const {user} = useAuth();
  const [message,setMessage] = useState(""), [busy,setBusy] = useState(false), [error,setError] = useState("");
  const [available,setAvailable] = useState<boolean|null>(null);
  const [messages,setMessages] = useState<{role:string;text:string;source?:string;generatedAt?:string}[]>([]);
  const input = useRef<HTMLTextAreaElement>(null), end = useRef<HTMLDivElement>(null);
  useEffect(() => {input.current?.focus();apiFetch("/workspace/assistant/status").then(response => setAvailable(response.data.available)).catch(e => {setAvailable(false);setError(e.message);});},[]);
  useEffect(() => {end.current?.scrollIntoView({block:"nearest"});},[messages,busy]);
  async function submit(event:React.FormEvent) {
    event.preventDefault();if(!message.trim() || busy || !available)return;
    const question=message.trim();setMessage("");setError("");setBusy(true);setMessages(previous => [...previous,{role:"user",text:question}]);
    try {const response=await apiFetch("/workspace/assistant",{method:"POST",body:JSON.stringify({message:question})});setMessages(previous => [...previous,{role:"assistant",text:response.data.answer,source:response.data.source,generatedAt:response.data.generatedAt}]);}
    catch(e:any) {setError(e.message);} finally {setBusy(false);input.current?.focus();}
  }
  const examples = user?.userType === "FINANCE" ? ["How much fee is pending?","What did we collect this month?"] : user?.userType === "TEACHER" ?   return <div className="fixed inset-0 z-[70] flex justify-end"><button onClick={onClose} aria-label="Close Acadivo AI" className="absolute inset-0 bg-black/20"/><section role="dialog" aria-modal="true" aria-label="Acadivo AI assistant" className="relative flex h-full w-full max-w-md flex-col border-l border-zinc-200 bg-white shadow-lg" onKeyDown={event => {if(event.key === "Tab") {const focusable=event.currentTarget.querySelectorAll<HTMLElement>('button:not(:disabled), textarea:not(:disabled)');const first=focusable[0],last=focusable[focusable.length-1];if(event.shiftKey && document.activeElement===first){event.preventDefault();last?.focus();}else if(!event.shiftKey && document.activeElement===last){event.preventDefault();first?.focus();}}}}>
    <header className="flex shrink-0 items-center gap-3 border-b p-5"><MessageSquare size={20} className="text-brand-600"/><div><h2 className="font-semibold">Acadivo AI</h2><p className="text-xs text-zinc-500">{user?.school?.name || "Platform workspace"}</p></div><button onClick={onClose} aria-label="Close assistant" className="ml-auto rounded p-2 hover:bg-zinc-100"><X size={18}/></button></header>
    <div className="min-h-0 flex-1 overflow-y-auto p-5" aria-live="polite">{!messages.length && <div className="space-y-4 py-6"><h3 className="text-xl font-semibold">Ask about your workspace.</h3><p className="text-sm leading-relaxed text-zinc-600">Get answers from the records your account can access. The assistant can read data; it cannot change school records.</p>{available === false ? <p className="rounded-lg border border-amber-200 bg-amber-50 p-4 text-sm text-amber-900">The AI provider is not configured or unavailable. Your existing CRM tools remain available.</p> : available === null ? <p className="text-sm text-zinc-500">Checking assistant availability…</p> : examples.map(example => <button key={example} onClick={() => {setMessage(example);input.current?.focus();}} className="block w-full rounded-lg border p-3 text-left text-sm hover:bg-zinc-50">{example}</button>)}</div>}{messages.map((item,index) => <article key={index} className={"mb-5 "+(item.role === "user" ? "ml-8 rounded-lg bg-zinc-100 p-3" : "border-b pb-5")}><p className="mb-2 text-[10px] font-semibold uppercase tracking-wider text-zinc-500">{item.role === "user" ?     <form onSubmit={submit} className="shrink-0 border-t p-4"><label htmlFor="assistant-question" className="sr-only">Ask Acadivo</label><div className="flex items-end gap-2 rounded-lg border p-2 focus-within:border-zinc-500"><textarea id="assistant-question" ref={input} value={message} onChange={e => setMessage(e.target.value)} maxLength={1200} disabled={!available || busy} placeholder="Ask about your school…" rows={2} className="min-w-0 flex-1 resize-none border-0 bg-white p-1 text-sm outline-none disabled:opacity-50"/><button type="submit" disabled={!available || busy || !message.trim()} aria-label="Send question" className="rounded-md bg-brand-600 p-2 text-white disabled:opacity-40"><ArrowUp size={18}/></button></div><p className="mt-2 text-[10px] leading-relaxed text-zinc-500">Answers may contain mistakes.   </section></div>;
}

import {useId} from "react";
export type ChartSeries={label:string;value:number}[];
export default function DataChart({title,data,kind="bar",unit="",description=""}:{title:string;data:ChartSeries;kind?:"bar"|"line"|"donut";unit?:string;description?:string}) {
  const id=useId();const rows=data.filter(row=>Number.isFinite(row.value));
  const total=rows.reduce((sum,row)=>sum+Math.max(0,row.value),0),max=Math.max(1,...rows.map(row=>row.value));
  const format=(value:number)=> (unit === "PKR" ? "PKR " : "")+value.toLocaleString(undefined,{maximumFractionDigits:1})+(unit === "%" ? "%" : "");
  const palette=["#E0000B","#111111","#71717a","#d4d4d8","#fca5a5"];
  const width=600,height=220,pad=30,step=(width-pad*2)/Math.max(1,rows.length-1);
  const points=rows.map((row,index)=>[pad+index*step,height-pad-row.value/max*(height-pad*2)]);
  let offset=0;
  return <section className="min-w-0 rounded-lg border border-zinc-200 bg-white p-5" aria-labelledby={id}><h3 id={id} className="text-sm font-semibold">{title}</h3>{description && <p className="mt-1 text-xs leading-5 text-zinc-500">{description}</p>}{!total ? <div className="flex min-h-40 items-center justify-center text-sm text-zinc-500">No recorded data available.</div> : kind === "donut" ? <div className="mt-6 flex flex-wrap items-center justify-center gap-6"><svg role="img" aria-label={title} viewBox="0 0 120 120" className="h-36 w-36 shrink-0"><title>{rows.map(row=>row.label+": "+format(row.value)).join("; ")}</title>{rows.map((row,index)=>{const length=Math.max(0,row.value)/total*100;const previous=offset;offset+=length;return <circle key={index} cx="60" cy="60" r="45" pathLength="100" fill="none" stroke={palette[index%palette.length]} strokeWidth="13" strokeDasharray={`${length} ${100-length}`} strokeDashoffset={-previous} transform="rotate(-90 60 60)"><title>{row.label}: {format(row.value)}</title></circle>;})}<text x="60" y="57" textAnchor="middle" fill="#111111" fontSize="14" fontWeight="600">{unit === "PKR" ? "PKR" : total.toLocaleString()}</text><text x="60" y="74" textAnchor="middle" fill="#71717a" fontSize="9">{unit === "PKR" ? format(total).replace("PKR ","") : "recorded entries"}</text></svg><ul className="min-w-40 space-y-3 text-xs">{rows.map((row,index)=><li key={index} className="flex items-center gap-2"><span className="h-2 w-2 rounded-full" style={{background:palette[index%palette.length]}}/><span className="flex-1 capitalize text-zinc-600">{row.label.toLowerCase().replaceAll("_"," ")}</span><strong className="font-semibold">{format(row.value)}</strong></li>)}</ul></div> : kind === "line" ? <div className="mt-4"><svg role="img" aria-label={title} viewBox={`0 0 ${width} ${height}`} className="w-full"><title>{rows.map(row=>row.label+": "+format(row.value)).join("; ")}</title>{[0,0.5,1].map(fraction=><g key={fraction}><line x1={pad} x2={width-pad} y1={height-pad-fraction*(height-pad*2)} y2={height-pad-fraction*(height-pad*2)} stroke="#e4e4e7"/><text x={pad} y={height-pad-fraction*(height-pad*2)-6} fontSize="10" fill="#71717a">{format(max*fraction)}</text></g>)}{rows.length > 1 && <polyline points={points.map(point=>point.join(",")).join(" ")} fill="none" stroke="#E0000B" strokeWidth="2.5"/>}{points.map((point,index)=><circle key={index} cx={point[0]} cy={point[1]} r="4" fill="#E0000B"><title>{rows[index].label}: {format(rows[index].value)}</title></circle>)}</svg><div className="flex justify-between text-[10px] text-zinc-500"><span>{rows[0]?.label}</span><span>{rows.at(-1)?.label}</span></div>{rows.length === 1 && <p className="mt-3 text-xs text-zinc-500">One recorded period; more data is needed to show a trend.</p>}</div> : <div className="mt-5 max-h-80 space-y-3 overflow-y-auto">{rows.map((row,index)=><div key={index} className="grid grid-cols-[minmax(70px,1fr)_2fr] items-center gap-3 text-xs"><span className="truncate text-zinc-600" title={row.label}>{row.label}</span><div><div className="mb-1 text-right font-medium">{format(row.value)}</div><div className="h-2 rounded-sm bg-zinc-100"><div className="h-full rounded-sm bg-brand-600" style={{width:Math.max(0,row.value)/max*100+"%"}}/></div></div></div>)}</div>}
    {total > 0 && <details className="mt-5 border-t pt-3"><summary className="cursor-pointer text-xs font-medium text-zinc-600">View source values</summary><div className="mt-3 overflow-x-auto"><table className="w-full text-left text-xs"><thead><tr><th className="p-2 font-medium">Record</th><th className="p-2 text-right font-medium">Value</th></tr></thead><tbody>{rows.map((row,index)=><tr key={index} className="border-t"><td className="p-2">{row.label}</td><td className="p-2 text-right">{format(row.value)}</td></tr>)}</tbody></table></div></details>}
  </section>;
}
          {!billingOnly && <button onClick={() => void openNotifications()} disabled={role==="PARENT"&&!parentChild} aria-label="Notifications" aria-expanded={menu === "notifications"} className="relative rounded-md p-2 hover:bg-zinc-100"><Bell size={18}/>{notifications?.unreadCount > 0 && <span className="absolute right-1 top-1 h-2 w-2 rounded-full bg-brand-600"/>}</button>}
          <button onClick={() => setMenu(menu === "profile" ? null : "profile")} aria-label="Account menu" aria-expanded={menu === "profile"} className="flex min-w-0 items-center gap-2 border-l border-zinc-200 pl-3"><span className="grid h-8 w-8 shrink-0 place-items-center rounded-full bg-zinc-100 text-xs font-semibold">{user?.name?.slice(0,2).toUpperCase()}</span><span className="hidden text-left sm:block"><span className="block max-w-36 truncate text-xs font-semibold">{user?.name}</span><span className="block text-[10px] text-zinc-500">{config.title}</span></span><ChevronDown size={14}/></button>
        </header>
        {menu === "profile" && <div className="absolute right-4 top-14 w-64 rounded-lg border border-zinc-200 bg-white p-2 shadow-lg"><div className="border-b px-3 py-3"><p className="text-sm font-semibold">{user?.name}</p><p className="break-all text-xs text-zinc-500">{user?.email}</p><p className="mt-1 text-xs text-zinc-500">{user?.school?.name || "Acadivo platform"}</p></div>{!billingOnly && <Link href={base+(links.some(item => item.path === "/profile") ? "/profile" : "/settings")} className="flex items-center gap-2 rounded px-3 py-2 text-sm hover:bg-zinc-50"><Settings size={15}/>Account & settings</Link>}<button onClick={()=>{setMenu(null);setPasswordOpen(true);}} className="block w-full rounded px-3 py-2 text-left text-sm hover:bg-zinc-50">Change password</button><button onClick={() => void logout()} className="flex w-full items-center gap-2 rounded px-3 py-2 text-sm text-red-700 hover:bg-red-50"><LogOut size={15}/>Sign out</button></div>}
        {menu === "notifications" && <div className="absolute right-4 top-14 max-h-[70vh] w-[min(360px,calc(100vw-32px))] overflow-auto rounded-lg border border-zinc-200 bg-white p-4 shadow-lg"><h2 className="text-sm font-semibold">Notifications</h2>{notificationError ? <p role="alert" className="mt-3 text-xs text-red-700">{notificationError}</p> : !notifications ? <p className="mt-3 text-xs text-zinc-500">Loading notifications…</p> : notifications.items.length ? notifications.items.map((item:any) => <article key={item.id} className="border-b py-3 last:border-0"><p className="text-sm font-medium">{item.title}</p><p className="mt-1 text-xs leading-relaxed text-zinc-600">{item.content}</p><p className="mt-2 text-[10px] text-zinc-400">{new Date(item.createdAt).toLocaleString()}</p>{item.canMarkRead && !item.isRead && <button onClick={() => void markRead(item.id)} className="mt-2 text-xs font-medium text-brand-600">Mark as read</button>}</article>) : <p className="mt-4 text-xs text-zinc-500">No notifications available.</p>}</div>}
      </div>
      <main ref={main} id="workspace-content" className="portal-content min-h-0 min-w-0 flex-1 overflow-y-auto overscroll-contain p-4 sm:p-6 lg:p-8"><div className="mx-auto max-w-[1440px]">{user?.isOrganizationOwner && !user.branchId && <p className="mb-4 rounded-lg border bg-white px-4 py-3 text-xs text-zinc-600">Viewing all branches. Choose a specific branch before creating or editing campus records.</p>}{billingOnly && !pathname.endsWith("/payments") && !pathname.endsWith("/subscription") ? <p className="text-sm text-zinc-500">Opening subscription settings…</p> : needsAttendanceSetup ? <AttendanceSetup mandatory onComplete={()=>void refreshUser()}/> : children}</div></main>
    </div>
    {passwordOpen && <ChangePassword onClose={()=>setPasswordOpen(false)}/>}
    <ScreenshotPrivacy/>
    {assistant && !billingOnly && role !== "PARENT" && <AcadivoAssistant onClose={() => setAssistant(false)}/>}
  </div>;
}

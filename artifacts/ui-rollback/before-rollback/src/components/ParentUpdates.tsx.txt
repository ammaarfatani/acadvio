"use client";
import {useEffect,useState} from "react";
import Link from "next/link";
import {apiFetch} from "@/lib/api";
import {useParent} from "@/lib/parent-context";
export default function ParentUpdates({href}:{href:string}){const {selected}=useParent();const [state,setState]=useState<any>({id:"",items:[],loading:true,error:""});useEffect(()=>{if(!selected)return;let live=true;setState({id:selected,items:[],loading:true,error:""});const load=()=>apiFetch("/workspace/notifications?studentId="+selected).then(r=>{if(live)setState({id:selected,items:r.data.items,loading:false,error:""});}).catch(e=>{if(live)setState({id:selected,items:[],loading:false,error:e.message});});void load();const timer=setInterval(()=>{if(document.visibilityState==="visible")void load();},30000);return()=>{live=false;clearInterval(timer);};},[selected]);return <section className="rounded-xl border bg-white p-5"><div className="flex items-center justify-between gap-4"><h2 className="font-semibold">Recent school updates</h2><Link href={href} className="text-sm text-red-600">View all</Link></div>{state.loading||state.id!==selected?<p role="status" className="mt-4 text-sm text-zinc-500">Loading updates...</p>:state.error?<p role="alert" className="mt-4 text-sm text-red-700">{state.error}</p>:state.items.length?state.items.slice(0,3).map((n:any)=><article className="mt-4 border-t pt-4" key={n.id}><p className="text-sm font-medium">{n.title}</p><p className="mt-1 line-clamp-2 text-sm text-zinc-500">{n.content}</p></article>):<p className="mt-4 text-sm text-zinc-500">No current school updates for this child.</p>}</section>;}

"use client";
import {useEffect,useRef,useState} from "react";
import {ArrowUp,ArrowUpRight,MessageSquare,X,RefreshCw,LockKeyhole} from "lucide-react";
import {apiFetch} from "@/lib/api";
import {useAuth} from "@/lib/auth-context";
type ConversationMessage={role:"user"|"assistant";text:string;source?:string;generatedAt:string};
export default function AcadivoAssistant({onClose}:{onClose:()=>void}){
 const {user}=useAuth();const [message,setMessage]=useState(""),[busy,setBusy]=useState(false),[error,setError]=useState(""),[available,setAvailable]=useState<boolean|null>(null),[messages,setMessages]=useState<ConversationMessage[]>([]),[failed,setFailed]=useState("");
 const input=useRef<HTMLTextAreaElement>(null),end=useRef<HTMLDivElement>(null),active=useRef(true),pending=useRef(false);
 async function checkAvailability(){setAvailable(null);setError("");try{const response=await apiFetch("/workspace/assistant/status");if(active.current)setAvailable(response.data.available);}catch(e:any){if(active.current){setAvailable(false);setError(e.message);}}}
 useEffect(()=>{active.current=true;const previous=document.activeElement as HTMLElement|null;input.current?.focus();void checkAvailability();return()=>{active.current=false;previous?.focus();};},[]);
 useEffect(()=>{end.current?.scrollIntoView({block:"nearest"});},[messages,busy]);
 async function send(question:string,retry=false){if(!question.trim()||pending.current||!available)return;pending.current=true;setBusy(true);setMessage("");setError("");setFailed("");if(!retry)setMessages(previous=>[...previous,{role:"user",text:question,generatedAt:new Date().toISOString()}]);try{const response=await apiFetch("/workspace/assistant",{method:"POST",body:JSON.stringify({message:question})});if(active.current)setMessages(previous=>[...previous,{role:"assistant",text:response.data.answer,source:response.data.source,generatedAt:response.data.generatedAt||new Date().toISOString()}]);}catch(e:any){if(active.current){setError(e.message);setFailed(question);}}finally{pending.current=false;if(active.current){setBusy(false);input.current?.focus();}}}
 const examples=user?.userType==="FINANCE"?["How much fee is pending?","What did we collect this month?"]:user?.userType==="TEACHER"?["Summarize my assigned classes.","What is my timetable today?"]:user?.userType==="SUPER_ADMIN"?["Summarize platform subscriptions.","What is this month's verified revenue?"]:["How many active students are there?","Give me an attendance summary."];
 return <div className="fixed inset-0 z-[70] flex justify-end"><button onClick={onClose} aria-label="Close Acadivo AI" className="absolute inset-0 bg-black/35 backdrop-blur-[2px]"/><section role="dialog" aria-modal="true" aria-label="Acadivo AI assistant" className="premium-assistant relative flex h-full w-full max-w-lg flex-col border-l border-zinc-200 shadow-2xl" onKeyDown={event=>{if(event.key==="Escape")onClose();if(event.key==="Tab"){const controls=event.currentTarget.querySelectorAll<HTMLElement>('button:not(:disabled),textarea:not(:disabled)');const first=controls[0],last=controls[controls.length-1];if(event.shiftKey&&document.activeElement===first){event.preventDefault();last?.focus();}else if(!event.shiftKey&&document.activeElement===last){event.preventDefault();first?.focus();}}}}>
 <header className="premium-assistant-header flex shrink-0 items-center gap-3"><span className="rounded-xl border border-white/20 bg-white/10 p-3"><MessageSquare size={22}/></span><div className="min-w-0"><p className="text-[10px] font-medium uppercase tracking-[.16em] text-red-300">Your workspace assistant</p><h2 className="mt-1 text-lg font-semibold">Acadivo AI</h2><p className="mt-1 truncate text-xs text-zinc-500">{user?.school?.name||"Platform workspace"}</p></div><button onClick={onClose} aria-label="Close assistant" className="ml-auto shrink-0 rounded-lg p-2"><X size={19}/></button></header>
 <div className="flex items-center gap-2 border-b border-zinc-200 bg-white px-6 py-3 text-xs text-zinc-500"><LockKeyhole size={13}/><span>Answers follow your account permissions</span></div>
 <div className="min-h-0 flex-1 overflow-y-auto px-5 py-6" role="log" aria-live="polite" aria-relevant="additions">{!messages.length&&<div className="space-y-5 py-4"><div className="assistant-intro-label"><span className="assistant-intro-rule" aria-hidden="true"/>WORKSPACE CONVERSATION</div><div><h3 className="text-2xl font-semibold tracking-tight">What would you like to understand?</h3><p className="mt-3 text-sm leading-6 text-zinc-600">Ask a question about the records your account can access. The assistant reads information; it does not edit your school records.</p></div>{available===false?<div className="rounded-xl border border-amber-200 bg-amber-50 p-4 text-sm leading-6 text-amber-900">The configured AI provider is unavailable.<button onClick={()=>void checkAvailability()} className="mt-3 flex items-center gap-2 font-medium"><RefreshCw size={14}/>Check again</button></div>:available===null?<p role="status" className="text-sm text-zinc-500">Connecting to your assistant…</p>:<div className="space-y-3"><p className="text-xs font-medium uppercase tracking-wider text-zinc-500">Start with a question</p>{examples.map(example=><button key={example} onClick={()=>{setMessage(example);input.current?.focus();}} className="premium-chat-prompt"><span>{example}</span><ArrowUpRight size={16} className="shrink-0 text-brand-600"/></button>)}</div>}</div>}
 {messages.map((item,index)=><article key={index} className={"mb-5 "+(item.role==="user"?"premium-chat-user":"premium-chat-answer")}><div className="mb-2 flex items-center justify-between gap-3 text-[11px] text-zinc-500"><span className="font-medium">{item.role==="user"?"You":"Acadivo AI"}</span><time dateTime={item.generatedAt}>{new Date(item.generatedAt).toLocaleTimeString([],{hour:"2-digit",minute:"2-digit"})}</time></div><p className="whitespace-pre-wrap break-words text-sm leading-7">{item.text}</p>{item.source&&<p className="mt-4 border-t pt-3 text-xs text-zinc-500">Source: {item.source}</p>}</article>)}
 {busy&&<div role="status" className="premium-chat-answer mb-5 flex items-center gap-3 text-sm text-zinc-500"><span className="premium-typing" aria-hidden="true"><i/><i/><i/></span>Checking your authorized records…</div>}{error&&<div role="alert" className="rounded-xl border border-red-200 bg-red-50 p-4 text-sm leading-6 text-red-800">{error}{failed&&<button disabled={busy} onClick={()=>void send(failed,true)} className="mt-3 flex items-center gap-2 font-medium"><RefreshCw size={14}/>Retry this question</button>}</div>}<div ref={end}/></div>
 <form onSubmit={event=>{event.preventDefault();void send(message.trim());}} className="shrink-0 border-t border-zinc-200 bg-white p-5"><label htmlFor="assistant-question" className="sr-only">Ask Acadivo</label><div className="premium-chat-composer flex items-end gap-3"><textarea id="assistant-question" ref={input} value={message} onChange={e=>setMessage(e.target.value)} onKeyDown={e=>{if(e.key==="Enter"&&!e.shiftKey){e.preventDefault();e.currentTarget.form?.requestSubmit();}}} maxLength={1200} disabled={!available||busy} placeholder="Ask about your workspace…" rows={2} className="min-w-0 flex-1 resize-none border-0 bg-transparent p-1 text-sm leading-6 outline-none disabled:opacity-50"/><button type="submit" disabled={!available||busy||!message.trim()} aria-label="Send question" className="rounded-xl bg-brand-600 p-3 text-white disabled:opacity-40"><ArrowUp size={19}/></button></div><p className="mt-3 text-[11px] leading-5 text-zinc-500">Enter to send · Shift + Enter for a new line. Check source records before making decisions.</p></form>
 </section></div>;
}

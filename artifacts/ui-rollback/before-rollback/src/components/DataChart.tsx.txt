"use client";
import {useId,useState} from "react";
import {EmptyRecords} from "./PremiumUI";
export type ChartSeries={label:string;value:number}[];
export default function DataChart({title,data,kind="bar",unit="",description=""}:{title:string;data:ChartSeries;kind?:"bar"|"line"|"donut";unit?:string;description?:string}){
 const id=useId(),[selected,setSelected]=useState<number|null>(null);const rows=data.filter(row=>Number.isFinite(row.value)),total=rows.reduce((sum,row)=>sum+Math.max(0,row.value),0),max=Math.max(1,...rows.map(row=>row.value)),min=Math.min(0,...rows.map(row=>row.value)),magnitude=Math.max(1,...rows.map(row=>Math.abs(row.value)));
 const format=(value:number)=>(unit==="PKR"?"PKR ":"")+value.toLocaleString(undefined,{maximumFractionDigits:1})+(unit==="%"?"%":"");
 const palette=["#e0000b","#090a0e","#8d8581","#ccc6be","#edaaa6"];
 const width=600,height=220,pad=30,step=(width-pad*2)/Math.max(1,rows.length-1),y=(value:number)=>height-pad-(value-min)/(max-min)*(height-pad*2),points=rows.map((row,index)=>[rows.length===1?width/2:pad+index*step,y(row.value)]);
 let offset=0;const chosen=selected===null?null:rows[selected];
 return <section className="premium-chart min-w-0" aria-labelledby={id}><div className="mb-2 flex items-start justify-between gap-3"><div><p className="mb-2 text-[10px] font-medium uppercase tracking-[.14em] text-zinc-500">Workspace analytics</p><h3 id={id} className="text-base font-semibold">{title}</h3></div><span className="mt-1 h-2 w-2 shrink-0 rounded-full bg-brand-600" aria-hidden="true"/></div>{description&&<p className="mt-1 text-xs leading-5 text-zinc-500">{description}</p>}
 {!rows.length?<div className="mt-5"><EmptyRecords message="No recorded data available for this view."/></div>:kind==="donut"&&total>0?<div className="mt-6 flex flex-wrap items-center justify-center gap-7"><svg role="img" aria-label={title} viewBox="0 0 120 120" className="h-40 w-40 shrink-0"><title>{rows.map(row=>row.label+": "+format(row.value)).join("; ")}</title><circle cx="60" cy="60" r="45" fill="none" stroke="#ece8e2" strokeWidth="13"/>{rows.map((row,index)=>{const length=Math.max(0,row.value)/total*100,previous=offset;offset+=length;return <circle key={index} cx="60" cy="60" r="45" pathLength="100" fill="none" stroke={palette[index%palette.length]} strokeWidth={selected===index?16:13} strokeDasharray={`${length} ${100-length}`} strokeDashoffset={-previous} transform="rotate(-90 60 60)" onMouseEnter={()=>setSelected(index)} onMouseLeave={()=>setSelected(null)}><title>{row.label+": "+format(row.value)}</title></circle>})}<text x="60" y="57" textAnchor="middle" fill="#090a0e" fontSize="14" fontWeight="600">{unit==="PKR"?"PKR":total.toLocaleString()}</text><text x="60" y="74" textAnchor="middle" fill="#71717a" fontSize="9">{unit==="PKR"?format(total).replace("PKR ",""):"recorded entries"}</text></svg><ul className="min-w-40 space-y-2 text-xs">{rows.map((row,index)=><li key={index}><button type="button" onMouseEnter={()=>setSelected(index)} onMouseLeave={()=>setSelected(null)} onFocus={()=>setSelected(index)} onBlur={()=>setSelected(null)} onClick={()=>setSelected(index)} className="flex w-full items-center gap-2 rounded-lg px-2 py-2 text-left hover:bg-zinc-100"><span className="h-2 w-2 shrink-0 rounded-full" style={{background:palette[index%palette.length]}}/><span className="flex-1 capitalize text-zinc-600">{row.label.toLowerCase().replaceAll("_"," ")}</span><strong className="font-semibold">{format(row.value)}</strong></button></li>)}</ul></div>:kind==="line"?<div className="mt-5"><svg role="img" aria-label={title} viewBox={`0 0 ${width} ${height}`} className="w-full"><title>{rows.map(row=>row.label+": "+format(row.value)).join("; ")}</title><defs><linearGradient id={id+"-area"} x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stopColor="#e0000b" stopOpacity=".16"/><stop offset="100%" stopColor="#e0000b" stopOpacity="0"/></linearGradient></defs>{[0,.5,1].map(f=><g key={f}><line x1={pad} x2={width-pad} y1={y(min+f*(max-min))} y2={y(min+f*(max-min))} stroke="#e3dfd8" strokeDasharray="3 5"/><text x={pad} y={y(min+f*(max-min))-7} fontSize="10" fill="#71717a">{format(min+f*(max-min))}</text></g>)}{rows.length>1&&<><polygon points={[...points, [points.at(-1)![0],y(0)],[points[0][0],y(0)]].map(point=>point.join(",")).join(" ")} fill={`url(#${id}-area)`}/><polyline points={points.map(point=>point.join(",")).join(" ")} fill="none" stroke="#e0000b" strokeWidth="3" strokeLinejoin="round"/></>}{points.map((point,index)=><g key={index}><circle cx={point[0]} cy={point[1]} r="12" fill="transparent" tabIndex={0} role="button" aria-label={rows[index].label+": "+format(rows[index].value)} onMouseEnter={()=>setSelected(index)} onMouseLeave={()=>setSelected(null)} onFocus={()=>setSelected(index)} onBlur={()=>setSelected(null)} onClick={()=>setSelected(index)} onKeyDown={e=>{if(e.key==="Enter"||e.key===" "){e.preventDefault();setSelected(index);}}}/><circle cx={point[0]} cy={point[1]} r={selected===index?6:4} fill="#e0000b" stroke="white" strokeWidth="2" pointerEvents="none"/></g>)}</svg><div className="flex justify-between gap-3 text-xs text-zinc-500"><span>{rows[0]?.label}</span><span>{rows.at(-1)?.label}</span></div>{rows.length===1&&<p className="mt-3 text-xs text-zinc-500">One recorded period; more data is needed to show a trend.</p>}</div>:<div className="mt-5 max-h-80 space-y-2 overflow-y-auto">{rows.map((row,index)=><button key={index} type="button" onMouseEnter={()=>setSelected(index)} onMouseLeave={()=>setSelected(null)} onFocus={()=>setSelected(index)} onBlur={()=>setSelected(null)} onClick={()=>setSelected(index)} className="grid w-full grid-cols-[minmax(70px,1fr)_2fr] items-center gap-3 rounded-lg p-2 text-left text-xs hover:bg-zinc-100"><span className="truncate text-zinc-600" title={row.label}>{row.label}</span><span><span className="mb-2 block text-right font-medium">{format(row.value)}</span><span className="block h-2.5 rounded-full bg-zinc-200/60"><span className="block h-full rounded-full bg-brand-600" style={{width:Math.abs(row.value)/magnitude*100+"%"}}/></span></span></button>)}</div>}
 {chosen&&<p role="status" className="mt-4 rounded-xl border border-red-100 bg-red-50 px-3 py-2 text-xs text-red-900">{chosen.label} · <strong>{format(chosen.value)}</strong></p>}
 {!!rows.length&&<details className="mt-5 border-t pt-4"><summary className="cursor-pointer text-xs font-medium text-zinc-600">View source values · {rows.length} records</summary><div className="mt-3 overflow-x-auto"><table className="w-full text-left text-xs"><thead><tr><th className="p-2 font-medium">Record</th><th className="p-2 text-right font-medium">Value</th></tr></thead><tbody>{rows.map((row,index)=><tr key={index} className="border-t"><td className="p-2">{row.label}</td><td className="p-2 text-right">{format(row.value)}</td></tr>)}</tbody></table></div></details>}
 </section>;
}

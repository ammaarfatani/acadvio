"use client";
import ModuleFrame from "@/components/ModuleFrame";

import {useCallback,useEffect,useState} from "react";
import {apiFetch,privateFile} from "@/lib/api";
export default function FeeVerification(){const [rows,setRows]=useState<any[]>([]),[error,setError]=useState(""),[loading,setLoading]=useState(true),[busy,setBusy]=useState(""),[reason,setReason]=useState<Record<string,string>>({}),[proof,setProof]=useState("");
 const load=useCallback(async()=>{try{setRows((await apiFetch("/finance/payment-requests")).data);setError("");}catch(e:any){setError(e.message);}finally{setLoading(false);}},[]);useEffect(()=>{void load();},[load]);useEffect(()=>()=>{if(proof)URL.revokeObjectURL(proof);},[proof]);
 async function review(id:string,status:string){setBusy(id);try{await apiFetch("/finance/payment-requests/"+id+"/review",{method:"POST",body:JSON.stringify({status,reason:reason[id]})});await load();}catch(e:any){setError(e.message);}finally{setBusy("");}}
 return <div className="space-y-6"><ModuleFrame><header className="flex flex-wrap justify-between gap-3 border-b pb-4"><div><h1 className="text-2xl font-black">Parent payment verification</h1><p className="mt-2 text-sm text-zinc-500">Approvals update the existing student fee ledger and create a receipt.</p></div><button onClick={()=>void load()} className="rounded-lg border bg-white px-4 py-2 text-sm">Refresh</button></header></ModuleFrame>{error&&<p role="alert" className="rounded-lg bg-red-50 p-3 text-sm text-red-700">{error}</p>}{loading?<p role="status">Loading requests...</p>:rows.length?rows.map(r=><article key={r._id} className="rounded-xl border bg-white p-5"><div className="flex flex-wrap justify-between gap-3"><h2 className="font-bold">{r.studentId?.name||"Student"} · {r.challanId?.challanNumber}</h2><span className="text-xs font-semibold">{r.status}</span></div><p className="mt-2 text-sm text-zinc-500">Parent: {r.parentId?.name} · {r.parentId?.email}</p><p className="mt-2 text-sm">PKR {r.amountPKR.toLocaleString()} · {r.paymentMethod} · {new Date(r.submittedAt).toLocaleString()}</p><p className="mt-1 text-sm">Reference: {r.reference||"Not provided"}</p>{r.rejectionReason&&<p className="mt-2 text-sm text-red-700">{r.rejectionReason}</p>}<button onClick={async()=>{try{setProof(URL.createObjectURL(await privateFile("/finance/payment-requests/"+r._id+"/proof")));}catch(e:any){setError(e.message);}}} className="mt-3 rounded-lg border px-3 py-2 text-sm">View screenshot</button>{r.status==="PENDING"&&<div className="mt-4 space-y-3"><label className="block text-sm">Rejection reason<input maxLength={1000} value={reason[r._id]||""} onChange={e=>setReason({...reason,[r._id]:e.target.value})} className="mt-1 block w-full rounded-lg border p-2"/></label><div className="flex gap-3"><button disabled={!!busy} onClick={()=>void review(r._id,"APPROVED")} className="rounded-lg bg-red-600 px-4 py-2 text-sm text-white">{busy===r._id?"Processing...":"Approve"}</button><button disabled={!!busy||!reason[r._id]?.trim()} onClick={()=>void review(r._id,"REJECTED")} className="rounded-lg border px-4 py-2 text-sm">Reject</button></div></div>}</article>):<p className="rounded-xl border border-dashed bg-white p-8 text-sm text-zinc-500">No parent payment requests.</p>}{proof&&<div role="dialog" aria-modal="true" aria-label="Payment screenshot" className="fixed inset-0 z-[100] grid place-items-center bg-black/40 p-4"><div className="max-h-[90vh] max-w-3xl overflow-auto rounded-xl bg-white p-4"><button onClick={()=>setProof("")} className="mb-3 rounded border px-3 py-2 text-sm">Close</button><img src={proof} alt="Uploaded payment screenshot" className="max-h-[75vh] max-w-full object-contain"/></div></div>}</div>;
}

"use client";
import ModuleFrame from "@/components/ModuleFrame";

import { useCallback, useEffect, useState } from "react";
import { apiFetch } from "@/lib/api";
export default function WorkspaceNotifications(){const [data,setData]=useState<any>(null),[loading,setLoading]=useState(true),[error,setError]=useState("");const load=useCallback(async()=>{setLoading(true);try{setData((await apiFetch("/workspace/notifications")).data);setError("");}catch(e:any){setError(e.message);}finally{setLoading(false);}},[]);useEffect(()=>{void load();},[load]);return <div className="space-y-5 pb-10"><ModuleFrame><header className="flex items-center justify-between border-b pb-4"><div><h1 className="text-2xl font-black">Notifications</h1><p className="mt-1 text-sm text-zinc-500">Your school notices and account updates.</p></div><button disabled={loading} onClick={()=>void load()} className="rounded-lg border bg-white px-4 py-2 text-sm">Refresh</button></header></ModuleFrame>{error&&<p role="alert" className="rounded bg-red-50 p-3 text-red-800">{error}</p>}{loading?<p role="status">Loading notifications…</p>:data?.items.length?data.items.map((n:any)=><article key={n.id} className="rounded-xl border bg-white p-5"><h2 className="font-semibold">{n.title}</h2><p className="mt-2 whitespace-pre-wrap text-sm text-zinc-600">{n.content}</p><p className="mt-2 text-xs text-zinc-500">{new Date(n.createdAt).toLocaleString()}</p>{n.canMarkRead&&!n.isRead&&<button className="mt-3 text-xs font-semibold text-red-600" onClick={async()=>{try{await apiFetch("/workspace/notifications/"+n.id+"/read",{method:"PATCH"});await load();}catch(e:any){setError(e.message);}}}>Mark as read</button>}</article>):<p className="rounded-xl border border-dashed bg-white p-10 text-center text-sm text-zinc-500">No notifications available.</p>}</div>;}

"use client";
import {useState} from "react";
export const field="w-full rounded-xl border border-zinc-200 bg-white px-3 py-3 text-sm";
export const button="rounded-lg bg-brand-600 px-4 py-2 text-sm font-medium text-white disabled:opacity-50";
export const card="rounded-2xl border border-zinc-200 bg-white p-6";
export const fmt=(n:any)=>(Number.isFinite(Number(n))?Number(n):0).toLocaleString(undefined,{minimumFractionDigits:2,maximumFractionDigits:2});
export const time=(s:any)=>s&&!Number.isNaN(new Date(s).getTime())?new Date(s).toLocaleTimeString("en-PK",{timeZone:"Asia/Karachi",hour:"2-digit",minute:"2-digit"}):"—";
export const roleLabel=(s:string)=>({TEACHER:"Teacher",FINANCE:"Finance",EXAM_STAFF:"Examination",RECEPTIONIST:"Reception",SCHOOL_ADMIN:"School administrator"}[s]||s?.replaceAll("_"," ")||"—");
export const currentMonth=()=>new Intl.DateTimeFormat("en-CA",{timeZone:"Asia/Karachi",year:"numeric",month:"2-digit"}).format(new Date());
export function Text({label,value,onChange,type="text",required=true,placeholder,help,min,max}:any){return <label className="grid content-start gap-1.5 text-xs font-semibold text-zinc-700">{label}<input required={required} type={type} min={min} max={max} placeholder={placeholder|| (type==="number"?"Enter amount or number":"Enter "+label.toLowerCase())} value={value??""} onChange={e=>onChange(e.target.value)} className={field+" min-h-11 focus:border-brand-500 focus:outline-none focus:ring-2 focus:ring-brand-100"}/>{help&&<span className="text-xs font-normal leading-relaxed text-zinc-500">{help}</span>}</label>;}
export function Select({label,value,onChange,options,empty="Select an option",required=false}:any){return <label className="grid content-start gap-1.5 text-xs font-semibold text-zinc-700">{label}<select required={required} className={field+" min-h-11"} value={value||""} onChange={e=>onChange(e.target.value)}><option value="">{empty}</option>{options.map((x:any)=><option key={typeof x==="string"?x:x.value} value={typeof x==="string"?x:x.value}>{typeof x==="string"?roleLabel(x):x.label}</option>)}</select></label>;}
export function Table({headers,rows,pageSize=25}:any){const [page,setPage]=useState(1);const pages=Math.max(1,Math.ceil(rows.length/pageSize));const active=Math.min(page,pages);return <div className="rounded-xl border border-zinc-200 bg-white"><div className="overflow-x-auto"><table className="w-full text-left text-sm"><thead className="bg-zinc-50 text-xs text-zinc-600"><tr>{headers.map((x:string)=><th className="whitespace-nowrap px-4 py-3 font-semibold" key={x}>{x}</th>)}</tr></thead><tbody>{rows.slice((active-1)*pageSize,active*pageSize).map((row:any[],i:number)=><tr className="border-t border-zinc-100 hover:bg-zinc-50/70" key={i}>{row.map((v:any,j:number)=><td className="px-4 py-3 align-top" key={j}>{v??"—"}</td>)}</tr>)}</tbody></table></div>{!rows.length&&<p className="p-8 text-center text-sm text-zinc-500">No records match your selection.</p>}{pages>1&&<div className="no-print flex items-center justify-between border-t p-3 text-xs text-zinc-600"><span>{rows.length} records · Page {active} of {pages}</span><div className="flex gap-3"><button disabled={active===1} className="rounded border px-3 py-2 disabled:opacity-40" onClick={()=>setPage(active-1)}>Previous</button><button disabled={active===pages} className="rounded border px-3 py-2 disabled:opacity-40" onClick={()=>setPage(active+1)}>Next</button></div></div>}</div>;}
export function Payslip({record,onClose}:any){const s=record.snapshot;return <div className="fixed inset-0 z-[80] overflow-y-auto bg-black/30 p-3 sm:p-8"><section id="workforce-payslip" className="mx-auto max-w-3xl rounded-xl bg-white p-6 sm:p-9"><div className="no-print flex justify-end gap-3"><button className={button} onClick={()=>window.print()}>Print / Save PDF</button><button onClick={onClose}>Close</button></div><h1 className="mt-4 text-2xl font-semibold">{s.school?.name||"School"}</h1><p className="text-sm text-zinc-500">{record.status==="PAID"?"Salary payment receipt":"Salary payslip"} · {record.month} · {record.reference}</p><div className="my-6 grid grid-cols-2 gap-4 text-sm"><p><b>{s.employee.name}</b><br/>{s.employee.employeeId}<br/>{roleLabel(s.employee.role)} · {s.employee.department||"Not configured"}</p><p>Status: <b>{record.status}</b><br/>Payment date: {record.paidAt?new Date(record.paidAt).toLocaleDateString():"Not recorded"}<br/>{record.paymentMethod} {record.paymentReference}</p></div><dl className="grid grid-cols-2 gap-x-6 gap-y-3 text-sm">{Object.entries({"Base salary":fmt(s.baseSalary)+" PKR ("+s.salaryBasis+")","Payable working days":s.payableDays,"Worked days":s.workedDays,"Paid leave":s.paidLeave,"Unpaid leave":s.unpaidLeave,"Late minutes":s.lateMinutes,"Gross salary":fmt(s.grossSalary),"Late deduction":fmt(s.lateDeduction),"Absence deduction":fmt(s.absenceDeduction),"Unpaid leave deduction":fmt(s.unpaidLeaveDeduction),"Other deduction":fmt(record.otherDeduction),"Net payable":fmt(Math.max(0,s.netSalary-record.otherDeduction))+" PKR"}).map(([k,v])=><div key={k} className="border-b py-2"><dt className="text-zinc-500">{k}</dt><dd className="font-semibold">{String(v)}</dd></div>)}</dl><p className="my-5 text-xs text-zinc-500">Daily salary = effective monthly salary ÷ {s.payableDays} payable days. Late deduction = late minutes × daily salary ÷ configured working minutes. Acadivo records the external payment; it does not transfer money.</p><div className="no-print"><h2 className="mb-3 mt-6 font-semibold">Attendance calculation details</h2><Table headers={["Date","Status","Source","Late minutes","Daily salary","Working minutes","Deduction"]} rows={s.days.map((d:any)=>[d.date,d.status,d.source,d.lateMinutes,fmt(d.dailySalary),d.workingMinutes,fmt(d.lateDeduction+d.absenceDeduction+d.unpaidDeduction)])}/></div><div className="mt-10 grid grid-cols-2 gap-12 text-xs text-zinc-600"><div className="border-t pt-3">Authorized by / signature<br/>{record.approvedByName||"School administration"}</div><div className="border-t pt-3">Employee acknowledgement</div></div></section></div>;}

export function PayrollTable({rows,onView,onAction,payOnly=false}:any){return <Table headers={["Employee / month","Department","Base PKR","Payable / worked","Paid / unpaid leave","Late min","Late / absence / unpaid deduction","Other","Gross","Net","Status","Actions"]} rows={rows.map((r:any)=>[r.snapshot.employee.name+" · "+r.month,r.snapshot.employee.department,fmt(r.snapshot.baseSalary),r.snapshot.payableDays+" / "+r.snapshot.workedDays,r.snapshot.paidLeave+" / "+r.snapshot.unpaidLeave,r.snapshot.lateMinutes,[r.snapshot.lateDeduction,r.snapshot.absenceDeduction,r.snapshot.unpaidLeaveDeduction].map(fmt).join(" / "),fmt(r.otherDeduction),fmt(r.snapshot.grossSalary),fmt(Math.max(0,r.snapshot.netSalary-r.otherDeduction)),r.status,<div key={r._id}><button className="text-brand-600" onClick={()=>onView(r)}>Details / Payslip</button>{onAction&&<div className="mt-2 flex gap-2">{(["CALCULATED","PENDING"].includes(r.status)?["ADJUST"]:[]).concat(r.status==="CALCULATED"?["SUBMIT"]:r.status==="PENDING"?["APPROVE"]:r.status==="APPROVED"?["PAY"]:[]).filter(a=>!payOnly||a==="PAY").map(a=><button className="rounded border p-1" key={a} onClick={()=>onAction(r,a)}>{({PAY:"Record payment",ADJUST:"Adjust deduction",SUBMIT:"Submit for approval",APPROVE:"Approve"} as any)[a]}</button>)}</div>}</div>])}/>;}

export function PayrollReport({rows,month,onClose}:any){const total=rows.reduce((n:number,r:any)=>n+Math.max(0,r.snapshot.netSalary-r.otherDeduction),0);return <div className="fixed inset-0 z-[80] overflow-y-auto bg-black/40 p-3 sm:p-8"><article id="payroll-report" className="mx-auto max-w-4xl rounded-xl bg-white p-6 sm:p-9"><div className="no-print flex justify-end gap-3"><button className={button} onClick={()=>window.print()}>Print / Save PDF</button><button onClick={onClose}>Close</button></div><header className="my-6 border-b-2 border-red-600 pb-5"><h1 className="text-2xl font-semibold">{rows[0]?.snapshot.school?.name||"School payroll"}</h1><h2 className="mt-2 font-semibold">Monthly payroll register · {month}</h2><p className="mt-2 text-sm text-zinc-600">{rows.length} employees · Net payable PKR {fmt(total)}</p><p className="text-xs text-zinc-500">Includes the employees matching your current filters. Payment status records external payments.</p></header><Table pageSize={Math.max(1,rows.length)} headers={["Employee / ID","Role / department","Gross PKR","Deductions PKR","Net PKR","Status"]} rows={rows.map((r:any)=>[<span key={r._id}>{r.snapshot.employee.name}<br/><small>{r.snapshot.employee.employeeId}</small></span>,roleLabel(r.snapshot.employee.role)+" · "+(r.snapshot.employee.department||"Not configured"),fmt(r.snapshot.grossSalary),fmt(r.snapshot.lateDeduction+r.snapshot.absenceDeduction+r.snapshot.unpaidLeaveDeduction+r.otherDeduction),fmt(Math.max(0,r.snapshot.netSalary-r.otherDeduction)),r.status])}/><footer className="mt-10 grid grid-cols-2 gap-12 text-xs"><p className="border-t pt-3">Prepared by / signature</p><p className="border-t pt-3">Authorized by / school stamp</p></footer></article></div>;}

"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { FormEvent, useCallback, useEffect, useState } from "react";
import { Layers, Plus, ShieldCheck, DoorOpen, Users, AlertCircle } from "lucide-react";
import { schoolAdminApi } from "@/lib/schoolAdminApi";

export default function SectionsPage() {
  const [sections, setSections] = useState<any[]>([]);
  const [classes, setClasses] = useState<any[]>([]);
  const [teachers, setTeachers] = useState<any[]>([]);
  const [assignments, setAssignments] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");

  const load = useCallback(async () => {
    setLoading(true);
    setError("");
    try {
      const [sectionRes, classRes, teacherRes, assignmentRes] = await Promise.all([
        schoolAdminApi.getSections(),
        schoolAdminApi.getClasses(),
        schoolAdminApi.getTeachers(),
        schoolAdminApi.getClassTeacherAssignments(),
      ]);
      setSections(sectionRes?.data || []);
      setClasses(classRes?.data || []);
      setTeachers(teacherRes?.data || []);
      setAssignments(assignmentRes?.data?.assignments || []);
    } catch (e: any) {
      setError(e?.message || "Unable to load sections.");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void load();
  }, [load]);

  async function create(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const formElement = event.currentTarget;
    const form = new FormData(formElement);
    setSaving(true);
    setError("");
    setNotice("");
    const linkedClass = classes.find((item) => item._id === form.get("classId"));
    try {
      await schoolAdminApi.createSection({
        classId: linkedClass?._id,
        className: linkedClass?.name,
        name: form.get("name"),
        roomNumber: form.get("roomNumber") || undefined,
        capacity: Number(form.get("capacity")) || undefined,
      });
      formElement.reset();
      setNotice(`Section '${form.get("name")}' created successfully.`);
      await load();
    } catch (e: any) {
      setError(e?.message || "Unable to create section.");
    } finally {
      setSaving(false);
    }
  }

  async function assign(section: any, teacherUserId: string) {
    if (!teacherUserId) return;
    setSaving(true);
    setError("");
    try {
      await schoolAdminApi.setClassTeacherAssignment({
        classId: section.classId,
        sectionId: section._id,
        teacherUserId,
      });
      setNotice("Class Teacher assigned for the current academic session.");
      await load();
    } catch (e: any) {
      setError(e?.message || "Unable to assign Class Teacher.");
    } finally {
      setSaving(false);
    }
  }

  return (
    <div className="space-y-6 pb-12 font-sans">
      {/* Header */}
      <ModuleFrame><div className="border-b border-zinc-200 pb-4 flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
        <div>
          <h1 className="text-2xl sm:text-3xl font-black text-zinc-950 flex items-center gap-2">
            <Layers className="w-6 h-6 text-red-600" />
            Class Sections & Class Teachers
          </h1>
          <p className="text-xs text-zinc-500 mt-0.5">
            Configure section divisions and designate section-specific Class Teachers
          </p>
        </div>

        <span className="text-xs font-bold px-3 py-1.5 bg-zinc-100 border border-zinc-200 rounded-xl text-zinc-700">
          Total Sections: {sections.length}
        </span>
      </div></ModuleFrame>

      {notice && (
        <div className="p-4 bg-emerald-50 border border-emerald-200 text-emerald-900 rounded-xl text-xs font-semibold flex items-center justify-between">
          <span>{notice}</span>
          <button onClick={() => setNotice("")} className="text-emerald-700 font-bold">Dismiss</button>
        </div>
      )}

      {error && (
        <div className="p-4 bg-red-50 border border-red-200 text-red-800 rounded-xl text-xs font-semibold flex items-center justify-between">
          <span>{error}</span>
          <button onClick={() => setError("")} className="text-red-700 font-bold">Dismiss</button>
        </div>
      )}

      {/* Form Card */}
      <div className="bg-white border border-zinc-200 rounded-2xl p-6 shadow-xs space-y-4">
        <div className="flex items-center gap-2 border-b border-zinc-100 pb-3">
          <Plus className="w-4 h-4 text-red-600" />
          <h2 className="text-sm font-bold text-zinc-900">Add New Section Division</h2>
        </div>

        <form onSubmit={create} className="grid gap-4 sm:grid-cols-2 lg:grid-cols-5 text-xs items-end">
          <label className="grid gap-1 font-bold text-zinc-800">
            Class *
            <select required name="classId" className="rounded-xl border border-zinc-200 bg-zinc-50 px-3 py-2 text-xs font-semibold text-zinc-900 focus:outline-none">
              <option value="">Select class</option>
              {classes.map((item) => (
                <option key={item._id} value={item._id}>
                  {item.name}
                </option>
              ))}
            </select>
          </label>

          <label className="grid gap-1 font-bold text-zinc-800">
            Section Name *
            <input required name="name" placeholder="e.g. Section A" className="rounded-xl border border-zinc-200 bg-zinc-50 px-3 py-2 text-xs font-medium focus:outline-none" />
          </label>

          <label className="grid gap-1 font-bold text-zinc-800">
            Room Number
            <input name="roomNumber" placeholder="e.g. Room 101" className="rounded-xl border border-zinc-200 bg-zinc-50 px-3 py-2 text-xs font-medium focus:outline-none" />
          </label>

          <label className="grid gap-1 font-bold text-zinc-800">
            Capacity Limit
            <input name="capacity" type="number" min="1" placeholder="40" className="rounded-xl border border-zinc-200 bg-zinc-50 px-3 py-2 text-xs font-mono focus:outline-none" />
          </label>

          <button
            disabled={saving || !classes.length}
            className="rounded-xl bg-zinc-950 hover:bg-zinc-800 px-4 py-2.5 text-xs font-bold text-white shadow-xs transition-colors disabled:opacity-50"
          >
            {saving ? "Creating…" : "Create Section"}
          </button>
        </form>
      </div>

      {/* Sections Grid */}
      {loading ? (
        <div className="p-8 text-center text-xs font-bold text-zinc-400 animate-pulse">
          Loading class sections directory…
        </div>
      ) : sections.length ? (
        <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
          {sections.map((section) => {
            const assignment = assignments.find((item) => String(item.sectionId?._id || item.sectionId) === section._id);
            const currentTeacherId = String(assignment?.teacherUserId?._id || assignment?.teacherUserId || "");
            const currentTeacher = teachers.find((t) => t._id === currentTeacherId);

            return (
              <article key={section._id} className="rounded-2xl border border-zinc-200 bg-white p-6 shadow-xs hover:border-zinc-300 transition-all space-y-4">
                <div className="flex items-start justify-between">
                  <div>
                    <h2 className="font-black text-zinc-950 text-base">{section.className} — {section.name}</h2>
                    <p className="text-xs text-zinc-500 font-mono mt-0.5">Room: {section.roomNumber || "Unassigned"}</p>
                  </div>
                  <Layers className="text-red-600 w-5 h-5 shrink-0" />
                </div>

                <div className="text-xs bg-zinc-50 p-3 rounded-xl border border-zinc-100 flex justify-between items-center">
                  <span className="text-zinc-500 font-medium">Capacity:</span>
                  <strong className="text-zinc-900 font-mono">{section.capacity || 40} Students</strong>
                </div>

                <div className="space-y-1.5 pt-2 border-t border-zinc-100">
                  <label className="block text-xs font-bold text-zinc-800 flex items-center justify-between">
                    <span>Class Teacher (Current Session)</span>
                    {currentTeacher && (
                      <span className="text-[10px] bg-emerald-100 text-emerald-800 px-2 py-0.5 rounded font-bold uppercase">
                        Assigned
                      </span>
                    )}
                  </label>
                  <select
                    value={currentTeacherId}
                    onChange={(event) => void assign(section, event.target.value)}
                    disabled={saving || !teachers.length}
                    className="w-full rounded-xl border border-zinc-200 bg-white px-3 py-2 text-xs font-bold text-zinc-900 focus:outline-none"
                  >
                    <option value="">No Class Teacher Assigned</option>
                    {teachers.map((teacher) => (
                      <option key={teacher._id} value={teacher._id}>
                        {teacher.name} ({teacher.email})
                      </option>
                    ))}
                  </select>
                </div>
              </article>
            );
          })}
        </div>
      ) : (
        <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-12 text-center space-y-2">
          <Layers className="w-10 h-10 text-zinc-300 mx-auto" />
          <h2 className="font-bold text-zinc-900 text-base">No Sections Configured</h2>
          <p className="text-xs text-zinc-500 max-w-sm mx-auto">
            Create a section after adding at least one class to assign Class Teachers.
          </p>
        </div>
      )}
    </div>
  );
}

"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useCallback, useEffect, useState } from "react";
import { teacherApi } from "@/lib/teacherApi";

export default function AdminTeacherLeaveRequestsPage() {
  const [reviewing,setReviewing]=useState<any>(null);const [comment,setComment]=useState("");
  const [requests, setRequests] = useState<any[]>([]); const [status, setStatus] = useState("ALL"); const [loading, setLoading] = useState(true); const [savingId, setSavingId] = useState(""); const [error, setError] = useState(""); const [notice, setNotice] = useState("");
  const load = useCallback(async () => { setLoading(true); setError(""); try { const response = await teacherApi.getSchoolLeaveRequests(status); setRequests(response.data || []); } catch (e: any) { setError(e.message || "Unable to load leave requests."); } finally { setLoading(false); } }, [status]);
  useEffect(() => { void load(); }, [load]);
  async function review(request: any, nextStatus: "APPROVED" | "REJECTED") { setSavingId(request._id); setError(""); setNotice(""); try { const response = await teacherApi.reviewLeaveRequest(request._id, nextStatus, comment || undefined); setNotice(response.message || `Leave request ${nextStatus.toLowerCase()}.`); await load();setReviewing(null); } catch (e: any) { setError(e.message || "Unable to review leave request."); } finally { setSavingId(""); } }
  return <div className="mx-auto max-w-7xl space-y-6 pb-12"><ModuleFrame><header className="border-b border-zinc-200 pb-5"><p className="text-xs font-semibold uppercase tracking-wider text-school-primary">Staff</p><h1 className="mt-1 text-3xl font-black">Staff Leave Requests</h1><p className="mt-1 text-sm text-zinc-500">Review leave requests from teachers, finance, examination and reception staff.</p></header></ModuleFrame>{error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{error}</p>}{notice && <p role="status" className="rounded-xl border border-emerald-200 bg-emerald-50 p-3 text-sm text-emerald-800">{notice}</p>}
    <label className="flex max-w-xs flex-col gap-1 text-xs font-semibold">Status<select value={status} onChange={(e) => setStatus(e.target.value)} className="rounded-lg border bg-white px-3 py-2"><option value="ALL">All requests</option><option value="PENDING">Pending</option><option value="APPROVED">Approved</option><option value="REJECTED">Rejected</option></select></label>
    {loading ? <p role="status" className="rounded-2xl bg-white p-6 text-sm text-zinc-500">Loading leave requests…</p> : requests.length ? <div className="overflow-x-auto rounded-2xl border border-zinc-200 bg-white"><table className="w-full min-w-[850px] text-left text-sm"><thead className="bg-zinc-50 text-xs uppercase text-zinc-500"><tr><th className="p-3">Employee</th><th className="p-3">Leave</th><th className="p-3">Dates</th><th className="p-3">Days</th><th className="p-3">Reason</th><th className="p-3">Status</th><th className="p-3">Review</th></tr></thead><tbody>{requests.map((request) => <tr key={request._id} className="border-t"><td className="p-3"><p className="font-semibold">{request.teacherName}</p><p className="text-xs text-zinc-500">{request.employeeId || ""}</p></td><td className="p-3">{request.leaveType}</td><td className="p-3">{new Date(request.startDate).toLocaleDateString()} – {new Date(request.endDate).toLocaleDateString()}</td><td className="p-3">{request.totalDays}</td><td className="max-w-xs p-3">{request.reason}</td><td className="p-3"><span className="rounded-full bg-zinc-100 px-2.5 py-1 text-xs font-semibold">{request.status}</span></td><td className="p-3">{request.status === "PENDING" ? <div className="flex gap-2"><button disabled={savingId === request._id} onClick={() => {setComment("");setReviewing({request,status:"APPROVED"});}} className="rounded-lg bg-school-primary px-3 py-2 text-xs font-semibold text-white disabled:opacity-50">Approve</button><button disabled={savingId === request._id} onClick={() => {setComment("");setReviewing({request,status:"REJECTED"});}} className="rounded-lg border border-zinc-200 px-3 py-2 text-xs font-semibold disabled:opacity-50">Reject</button></div> : <p className="text-xs text-zinc-600">{request.reviewComment || "No comment"}<span className="mt-1 block text-zinc-400">{request.reviewedByUserName || ""}</span></p>}</td></tr>)}</tbody></table></div> : <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-10 text-center"><h2 className="font-bold">No leave requests found</h2><p className="mt-1 text-sm text-zinc-500">New staff requests will appear here.</p></div>}{reviewing&&<div className="fixed inset-0 z-[70] overflow-y-auto bg-black/40 p-4 sm:p-10"><form onSubmit={e=>{e.preventDefault();void review(reviewing.request,reviewing.status);}} className="mx-auto max-w-lg space-y-4 rounded-xl border bg-white p-6"><h2 className="text-xl font-semibold">{reviewing.status==="APPROVED"?"Approve leave":"Reject leave"}</h2><p className="text-sm text-zinc-600">{reviewing.request.teacherName} · {reviewing.request.leaveType}<br/>{String(reviewing.request.startDate).slice(0,10)} – {String(reviewing.request.endDate).slice(0,10)}</p>{error&&<p role="alert" className="text-sm text-red-700">{error}</p>}<label className="block text-sm font-medium">Review comment (optional)<textarea maxLength={1000} value={comment} onChange={e=>setComment(e.target.value)} placeholder="Explain your decision to the employee" className="mt-2 min-h-28 w-full rounded-lg border p-3 text-sm"/></label><div className="flex justify-end gap-3"><button type="button" disabled={!!savingId} onClick={()=>setReviewing(null)} className="rounded-lg border px-4 py-2 text-sm">Cancel</button><button disabled={!!savingId} className="rounded-lg bg-red-600 px-4 py-2 text-sm text-white">{savingId?"Saving…":"Confirm decision"}</button></div></form></div>}</div>;
}

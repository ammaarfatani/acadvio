"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useCallback, useEffect, useState } from "react";
import { schoolAdminApi } from "@/lib/schoolAdminApi";

export default function ExpensesPage() {
  const [expenses, setExpenses] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [category, setCategory] = useState("ALL");

  const load = useCallback(async () => {
    setLoading(true);
    setError("");
    try {
      const res = await schoolAdminApi.getExpenses();
      setExpenses(res?.data || []);
    } catch (e: any) {
      setError(e?.message || "Unable to load expenses.");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void load();
  }, [load]);

  const filtered = category === "ALL" ? expenses : expenses.filter((item) => item.category === category);
  const totalAmount = filtered.filter(item => !item.isVoided).reduce((sum, item) => sum + Number(item.amountPKR || 0), 0);

  return (
    <div className="mx-auto max-w-6xl space-y-6 pb-12">
      <ModuleFrame><header className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b border-zinc-200 pb-4">
        <div>
          <h1 className="text-2xl font-black text-zinc-950">Expense Audit History</h1>
          <p className="mt-1 text-xs text-zinc-500">Monitor school operational expenses recorded by the Finance & Accounts department.</p>
        </div>

        <div className="text-right">
          <span className="text-[10px] uppercase font-bold text-zinc-500 block">Category Total</span>
          <span className="text-lg font-black text-red-600 font-mono">PKR {totalAmount.toLocaleString()}</span>
        </div>
      </header></ModuleFrame>

      {error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-xs font-semibold text-red-700">{error}</p>}

      <div className="flex items-center justify-between rounded-xl border border-zinc-200 bg-white p-3 shadow-xs">
        <h2 className="text-xs font-bold text-zinc-950 uppercase tracking-wider">Expense Vouchers Log</h2>
        <select
          aria-label="Filter expenses by category"
          value={category}
          onChange={(e) => setCategory(e.target.value)}
          className="rounded-lg border border-zinc-300 px-3 py-1.5 text-xs font-bold bg-zinc-50"
        >
          <option value="ALL">All categories</option>
          {["Salary", "Rent", "Electricity", "Stationery", "Maintenance", "Transport", "Other"].map((value) => (
            <option key={value} value={value}>{value}</option>
          ))}
        </select>
      </div>

      {loading ? (
        <p role="status" className="rounded-xl bg-white p-8 text-center text-xs text-zinc-500 border border-zinc-200">Loading expenses...</p>
      ) : filtered.length ? (
        <div className="overflow-x-auto rounded-2xl border border-zinc-200 bg-white shadow-xs">
          <table className="w-full min-w-[650px] text-left text-xs">
            <thead className="bg-zinc-950 text-white text-[10px] uppercase font-bold">
              <tr>
                <th className="p-3.5">Expense Title</th>
                <th className="p-3.5">Category</th>
                <th className="p-3.5 text-center">Date</th>
                <th className="p-3.5 text-center">Payment Method</th>
                <th className="p-3.5 text-right">Amount (PKR)</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-zinc-200">
              {filtered.map((item) => (
                <tr key={item._id} className="hover:bg-zinc-50">
                  <td className="p-3.5">
                    <span className="font-bold text-zinc-950 block">{item.title}</span>
                    <span className="text-[10px] text-zinc-500">{item.description}</span>
                  </td>
                  <td className="p-3.5">
                    <span className="bg-zinc-100 text-zinc-800 text-[10px] font-bold px-2 py-0.5 rounded">{item.category}</span>
                  </td>
                  <td className="p-3.5 text-center font-mono text-zinc-700 text-[11px]">{new Date(item.expenseDate).toLocaleDateString()}</td>
                  <td className="p-3.5 text-center font-semibold text-zinc-800">{item.paymentMethod}</td>
                  <td className="p-3.5 text-right font-mono font-bold text-red-600">PKR {Number(item.amountPKR || 0).toLocaleString()}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      ) : (
        <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-10 text-center space-y-1">
          <h3 className="font-bold text-zinc-950 text-xs">No recorded expenses found</h3>
          <p className="text-xs text-zinc-500">Expenses logged by Finance department will appear here for School Admin review.</p>
        </div>
      )}
    </div>
  );
}

"use client";
import {WorkspaceWelcome,MetricCard} from "@/components/PremiumUI";
import OrganizationWorkspace from "@/components/OrganizationWorkspace";
import WorkspaceAnalytics from "@/components/WorkspaceAnalytics";

import { useCallback, useEffect, useState } from "react";
import Link from "next/link";
import { BookOpen, CalendarDays, GraduationCap, Users } from "lucide-react";
import { schoolAdminApi } from "@/lib/schoolAdminApi";
import { useAuth } from "@/lib/auth-context";

type DashboardData = { school: { name: string; status: string; plan: string }; metrics: Record<string, any> };
const card = "rounded-2xl border border-zinc-200 bg-white p-5 shadow-sm";
export default function SchoolAdminDashboard() { const { user } = useAuth(); return user?.isOrganizationOwner ? <OrganizationWorkspace/> : <SingleSchoolDashboard/>; }
function SingleSchoolDashboard() {
  const { user } = useAuth(); const portalBase = user?.school?.slug ? `/${user.school.slug}/admin` : "/school-admin";
  const [data, setData] = useState<DashboardData | null>(null); const [error, setError] = useState(""); const [loading, setLoading] = useState(true);
  const load = useCallback(async () => { setLoading(true); try { const response = await schoolAdminApi.getDashboardMetrics(); setData(response.data); setError(""); } catch (e: any) { setError(e.message || "Could not load school dashboard."); } finally { setLoading(false); } }, []);
  useEffect(() => { void load(); }, [load]);
  const m = data?.metrics; const distribution = m?.studentDistribution || []; const maxClassCount = Math.max(1, ...distribution.map((row: any) => row.count));
  const primaryCards = [["Students", m?.totalStudents ?? 0, Users, "students"], ["Teachers", m?.totalTeachers ?? 0, GraduationCap, "teachers"], ["Staff users", m?.totalStaff ?? 0, Users, "staff"], ["Classes", m?.totalClasses ?? 0, BookOpen, "academics/classes"]] as const;
  const visibleCards = primaryCards.filter(([, , , path]) => path !== "staff" || user?.features?.includes("STAFF_ROLES"));
  return <div className="mx-auto max-w-7xl space-y-7 pb-12 text-zinc-900">
    <WorkspaceWelcome eyebrow={data?.school?.name||user?.school?.name||"School workspace"} title={"Welcome back, "+(user?.name?.split(" ")[0]||"Administrator")+"."} description="Your school’s people, attendance and academic operations — in one clear overview." context={m?.currentSessionName||"No active academic session"} actions={<><button onClick={()=>void load()} disabled={loading} className="rounded-xl border bg-white px-4 py-2.5 text-sm font-medium disabled:opacity-60">{loading?"Refreshing…":"Refresh overview"}</button>{user?.permissions?.includes("students.create")&&<Link href={`${portalBase}/students`} className="rounded-xl bg-brand-600 px-4 py-2.5 text-sm font-medium text-white">Student directory</Link>}</>}/>
    {error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{error}</p>}
    {loading && !data ? <div role="status" className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">{visibleCards.map(([label]) => <div key={label} className={`${card} h-28 animate-pulse bg-zinc-100`} />)}</div> : <>
      <section className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">{visibleCards.map(([label,value,Icon,path],i)=><Link key={label} href={`${portalBase}/${path}`} className="block"><MetricCard label={label} value={value} icon={Icon} tone={i===0?"brand":i===1?"ink":"neutral"}/></Link>)}</section>
      {m && !m.totalStudents && !m.totalTeachers && !m.totalClasses && !m.totalSubjects ? <section className={`${card} border-dashed p-8`}><h2 className="text-xl font-bold">Your school isn’t configured yet</h2><p className="mt-1 text-sm text-zinc-500">Start with a session and class, then add teachers and students. Dashboard totals will update as records are created.</p><div className="mt-5 flex flex-wrap gap-2">{[["Create academic session","academics/sessions"],["Create class","academics/classes"],["Add teacher","teachers"],["Add student","students"]].map(([label,path]) => <Link key={path} href={`${portalBase}/${path}`} className="rounded-xl bg-school-primary px-4 py-2.5 text-sm font-bold text-white hover:opacity-90">{label}</Link>)}</div></section> : <>
        <section className="grid gap-4 lg:grid-cols-2">
          <article className={card}><div className="flex items-start justify-between"><div><h2 className="text-base font-bold">Student attendance today</h2><p className="mt-1 text-sm text-zinc-500">From submitted class registers</p></div><Link href={`${portalBase}/attendance/students`} className="text-sm font-semibold text-school-primary">View report</Link></div><div className="mt-5 flex items-end gap-3"><p className="text-4xl font-black">{m?.todayAttendancePercent ?? 0}%</p><p className="pb-1 text-sm text-zinc-500">{m?.studentsPresentToday || 0} present · {m?.studentsAbsentToday || 0} absent · {m?.studentsLateToday || 0} late</p></div><div className="mt-4 h-2.5 overflow-hidden rounded-full bg-zinc-100"><div className="h-full rounded-full bg-school-primary transition-all" style={{ width: `${Math.min(100, m?.todayAttendancePercent || 0)}%` }} /></div></article>
          <article className={card}><div className="flex items-start justify-between"><div><h2 className="text-base font-bold">Teacher attendance today</h2><p className="mt-1 text-sm text-zinc-500">Actual teacher attendance records</p></div><Link href={`${portalBase}/attendance/teachers`} className="text-sm font-semibold text-school-primary">View report</Link></div><div className="mt-5 grid grid-cols-3 gap-3">{[["Present",m?.teachersPresentToday||0],["Absent",m?.teachersAbsentToday||0],["Records",m?.teachersWithAttendanceToday||0]].map(([label,value]) => <div key={String(label)} className="rounded-xl bg-zinc-50 p-3"><p className="text-xs text-zinc-500">{label}</p><p className="mt-1 text-2xl font-bold">{value}</p></div>)}</div></article>
        </section>
        <section className="grid gap-4 lg:grid-cols-2"><article className={card}><div className="flex items-center justify-between"><div><h2 className="font-bold">Students by class</h2><p className="mt-1 text-sm text-zinc-500">Active enrollment distribution</p></div><Link href={`${portalBase}/students`} className="text-sm font-semibold text-school-primary">Directory</Link></div>{distribution.length ? <div className="mt-5 space-y-4">{distribution.map((row: any) => <div key={row.className}><div className="mb-1.5 flex justify-between text-sm"><span>{row.className || "Unassigned class"}</span><span className="font-semibold">{row.count}</span></div><div className="h-2 overflow-hidden rounded-full bg-zinc-100"><div className="h-full rounded-full bg-school-primary" style={{ width: `${Math.max(2, (row.count / maxClassCount) * 100)}%` }} /></div></div>)}</div> : <p className="mt-5 rounded-xl border border-dashed border-zinc-200 p-6 text-center text-sm text-zinc-500">No data available yet.</p>}</article>
          <article className={card}><div className="flex items-center justify-between"><div><h2 className="font-bold">Academic setup</h2><p className="mt-1 text-sm text-zinc-500">Current configuration for this school</p></div><span className="rounded-xl bg-school-primary/10 p-2 text-school-primary"><BookOpen size={18} /></span></div><dl className="mt-5 divide-y divide-zinc-100">{[["Active session",m?.currentSessionName || "Not set"],["Classes",m?.totalClasses || 0],["Sections",m?.totalSections || 0],["Subjects",m?.totalSubjects || 0],["Timetable entries",m?.timetableSlots || 0]].map(([label,value]) => <div key={String(label)} className="flex justify-between py-3 text-sm"><dt className="text-zinc-500">{label}</dt><dd className="font-semibold">{value}</dd></div>)}</dl></article></section>
      </>}
      <p className="text-xs text-zinc-400">Dashboard figures are calculated from records stored for this school.</p>
    </>}
  <WorkspaceAnalytics/>
</div>;
}

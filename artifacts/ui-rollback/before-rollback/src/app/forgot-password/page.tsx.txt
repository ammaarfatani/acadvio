"use client";
import {useState} from "react";import {apiFetch} from "@/lib/api";
export default function Page(){const [email,setEmail]=useState(""),[password,setPassword]=useState(""),[confirm,setConfirm]=useState(""),[message,setMessage]=useState(""),[busy,setBusy]=useState(false);const reset=false;
return <main className="mx-auto max-w-md px-4 py-16"><h1 className="text-2xl font-black">{reset?"Reset password":"Forgot password"}</h1><form className="mt-6 space-y-4 rounded-xl border bg-white p-6" onSubmit={async e=>{e.preventDefault();setBusy(true);try{if(reset&&password!==confirm)throw new Error("Passwords must match.");const token=new URLSearchParams(window.location.search).get("token");const result=await apiFetch(reset?"/auth/reset-password":"/auth/forgot-password",{method:"POST",body:JSON.stringify(reset?{token,newPassword:password}:{email})});setMessage(result.message);}catch(e:any){setMessage(e.message);}finally{setBusy(false);}}}>{reset?<><label className="block text-sm">New password<input type="password" autoComplete="new-password" minLength={10} maxLength={72} required value={password} onChange={e=>setPassword(e.target.value)} className="mt-1 block w-full rounded-lg border p-2"/></label><label className="block text-sm">Confirm password<input type="password" required value={confirm} onChange={e=>setConfirm(e.target.value)} className="mt-1 block w-full rounded-lg border p-2"/></label></>:<label className="block text-sm">Email<input type="email" required value={email} onChange={e=>setEmail(e.target.value)} className="mt-1 block w-full rounded-lg border p-2"/></label>}{message&&<p role="status" className="text-sm">{message}</p>}<button disabled={busy} className="rounded-lg bg-red-600 px-4 py-2 text-white">{busy?"Submitting...":reset?"Reset password":"Send instructions"}</button><a href="/login" className="ml-3 text-sm">Sign in</a></form></main>;}

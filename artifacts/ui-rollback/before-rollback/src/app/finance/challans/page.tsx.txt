"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useEffect, useState } from "react";
import { FileSpreadsheet, Printer, Plus, CheckCircle2, RefreshCw } from "lucide-react";
import { financeApi } from "@/lib/financeApi";
import { useAuth } from "@/lib/auth-context";

export default function BankChallansPage() {
  const { user } = useAuth();
  const schoolName = user?.school?.name || "School ERP";

  const [classes, setClasses] = useState<Array<{ _id: string; name: string }>>([]);
  const [selectedClassId, setSelectedClassId] = useState("");
  const [selectedMonth, setSelectedMonth] = useState("October 2026");
  const [dueDate, setDueDate] = useState("2026-10-10");
  const [lateFeeFine, setLateFeeFine] = useState(500);

  const [generating, setGenerating] = useState(false);
  const [notice, setNotice] = useState<string | null>(null);
  const [loadError, setLoadError] = useState("");

  const [allChallans, setAllChallans] = useState<any[]>([]);
  const [selectedChallanIndex, setSelectedChallanIndex] = useState(0);

  useEffect(() => {
    financeApi.getClasses()
      .then((res) => {
        const clsList = Array.isArray(res?.data) ? res.data : [];
        setClasses(clsList);
        if (clsList.length > 0) {
          setSelectedClassId(clsList[0]._id);
        }
      })
      .catch(() => {});

    financeApi.getChallans()
      .then((res) => {
        if (res?.data && Array.isArray(res.data)) {
          setAllChallans(res.data);
        }
      })
      .catch(() => {});
  }, []);

  const handleBulkGenerate = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!selectedClassId) return;

    setGenerating(true);
    setLoadError("");

    try {
      const targetClassName = classes.find(c => c._id === selectedClassId)?.name || "selected class";
      const res = await financeApi.generateBulkChallans({
        classId: selectedClassId,
        className: targetClassName,
        month: selectedMonth,
        dueDate: dueDate,
        lateFeeFine,
      });

      // Reload challans
      const updatedChallans = await financeApi.getChallans();
      if (updatedChallans?.data && Array.isArray(updatedChallans.data)) {
        setAllChallans(updatedChallans.data);
      }

      setNotice(res?.message || `Bulk Fee Challans for ${targetClassName} (${selectedMonth}) generated successfully!`);
      setTimeout(() => setNotice(null), 4000);
    } catch (err: any) {
      setLoadError(err?.message || "Unable to generate batch fee challans.");
    } finally {
      setGenerating(false);
    }
  };

  const activeChallan = allChallans[selectedChallanIndex] || null;

  const currentChallanDoc = activeChallan ? {
    challanNo: activeChallan.challanNumber,
    studentName: activeChallan.studentName,
    rollNo: activeChallan.rollNumber,
    admissionNo: activeChallan.admissionNumber || "—",
    className: activeChallan.className,
    month: activeChallan.month,
    dueDate: new Date(activeChallan.dueDate).toLocaleDateString(),
    bankName: "Meezan Bank / HBL Counter Branch",
    accountNo: "0102-01048192-01",
    breakdown: Object.entries(activeChallan.feeBreakdown || {})
      .filter(([key, val]) => key !== "discount" && Number(val) > 0)
      .map(([key, val]) => ({
        title: key.replace(/([A-Z])/g, " $1").replace(/^./, str => str.toUpperCase()),
        amount: Number(val),
      })),
    netTotal: activeChallan.totalAmountPKR || 0,
    lateFeeFine: activeChallan.feeBreakdown?.lateFeeFine || 500,
  } : {
    challanNo: "CH-2026-PREVIEW-001",
    studentName: "Sample Student Record",
    rollNo: "1001",
    admissionNo: "ADM-2026-1001",
    className: "Class 10-A",
    month: selectedMonth,
    dueDate: dueDate,
    bankName: "Meezan Bank / HBL Counter Branch",
    accountNo: "0102-01048192-01",
    breakdown: [
      { title: "Monthly Tuition Fee", amount: 12000 },
      { title: "Exam Fee", amount: 1500 },
      { title: "Computer & IT Fund", amount: 1000 },
      { title: "Transport Fund", amount: 3000 },
      { title: "Library Fund", amount: 500 },
    ],
    netTotal: 18000,
    lateFeeFine: 500,
  };

  return (
    <div className="space-y-6 pb-12">
      {/* Header Toolbar */}
      <ModuleFrame><div className="print:hidden flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b border-zinc-200 pb-4">
        <div>
          <h1 className="text-2xl font-black text-zinc-950 flex items-center gap-2">
            <FileSpreadsheet className="h-6 w-6 text-red-600" />
            Bank Fee Challan Generator & 3-Copy Printer
          </h1>
          <p className="text-xs text-zinc-500">Bulk monthly challan batch generation with Student, Bank, and School copies</p>
        </div>

        <button
          onClick={() => window.print()}
          className="flex items-center gap-2 px-4 py-2 text-xs font-bold bg-red-600 hover:bg-red-700 text-white rounded-lg transition-colors shadow-xs shrink-0"
        >
          <Printer className="h-4 w-4" />
          Print 3-Copy Bank Challan
        </button>
      </div></ModuleFrame>

      {loadError && <p role="alert" className="print:hidden rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{loadError}</p>}
      {notice && (
        <div className="print:hidden bg-emerald-50 border border-emerald-200 text-emerald-800 px-4 py-3 rounded-xl text-xs font-medium flex items-center gap-2">
          <CheckCircle2 className="h-4 w-4 text-emerald-600 shrink-0" />
          <span>{notice}</span>
        </div>
      )}

      {/* Bulk Generator Card */}
      <div className="print:hidden bg-white border border-zinc-200 p-5 rounded-2xl shadow-xs space-y-4">
        <h2 className="text-sm font-bold text-zinc-950 border-b border-zinc-100 pb-2">
          Bulk Monthly Batch Challan Generator
        </h2>

        <form onSubmit={handleBulkGenerate} className="grid grid-cols-1 sm:grid-cols-4 gap-4 text-xs">
          <div>
            <label className="block font-bold text-zinc-700 uppercase mb-1">Target Class</label>
            <select
              value={selectedClassId}
              onChange={(e) => setSelectedClassId(e.target.value)}
              className="w-full border border-zinc-300 rounded-lg p-2.5 bg-zinc-50 focus:outline-none font-bold"
            >
              <option value="">Select Class</option>
              {classes.map((cls) => (
                <option key={cls._id} value={cls._id}>{cls.name}</option>
              ))}
            </select>
          </div>

          <div>
            <label className="block font-bold text-zinc-700 uppercase mb-1">Fee Month</label>
            <select
              value={selectedMonth}
              onChange={(e) => setSelectedMonth(e.target.value)}
              className="w-full border border-zinc-300 rounded-lg p-2.5 bg-zinc-50 focus:outline-none font-bold text-red-600"
            >
              <option>October 2026</option>
              <option>November 2026</option>
              <option>December 2026</option>
              <option>January 2027</option>
              <option>February 2027</option>
              <option>March 2027</option>
            </select>
          </div>

          <div>
            <label className="block font-bold text-zinc-700 uppercase mb-1">Due Date</label>
            <input
              type="date"
              value={dueDate}
              onChange={(e) => setDueDate(e.target.value)}
              className="w-full border border-zinc-300 rounded-lg p-2.5 font-mono focus:outline-none"
            />
          </div>

          <div className="flex items-end">
            <button
              type="submit"
              disabled={generating || !selectedClassId}
              className="w-full py-2.5 font-bold bg-red-600 hover:bg-red-700 text-white rounded-lg transition-colors shadow-xs flex items-center justify-center gap-1.5 disabled:opacity-50"
            >
              {generating ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Plus className="h-4 w-4" />}
              {generating ? "Generating..." : "Generate Batch Challans"}
            </button>
          </div>
        </form>
      </div>

      {/* Select Recorded Challan Selector */}
      {allChallans.length > 0 && (
        <div className="print:hidden bg-white border border-zinc-200 p-4 rounded-xl flex items-center justify-between gap-4 text-xs">
          <span className="font-bold text-zinc-950">Select Student Challan to Preview / Print ({allChallans.length} Challans in DB):</span>
          <select
            value={selectedChallanIndex}
            onChange={(e) => setSelectedChallanIndex(Number(e.target.value))}
            className="border border-zinc-300 rounded-lg px-3 py-1.5 font-mono font-bold bg-zinc-50"
          >
            {allChallans.map((c, idx) => (
              <option key={c._id || idx} value={idx}>
                {c.challanNumber} — {c.studentName} ({c.className} - {c.month})
              </option>
            ))}
          </select>
        </div>
      )}

      {/* 3-Copy Bank Challan Printable Sheet */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 bg-white border border-zinc-400 p-4 font-sans text-xs text-zinc-950 print:border-0 print:p-0">
        
        {["STUDENT COPY", "BANK COPY", "SCHOOL COPY"].map((copyType) => (
          <div key={copyType} className="border-2 border-zinc-900 p-4 space-y-3 bg-white">
            <div className="text-center border-b border-zinc-900 pb-2">
              <span className="font-black text-sm uppercase block">{schoolName}</span>
              <span className="text-[10px] font-bold text-red-600 uppercase block">{copyType}</span>
              <span className="text-[9px] text-zinc-500 font-mono block">{currentChallanDoc.bankName}</span>
              <span className="text-[9px] font-mono text-zinc-600 block">A/C: {currentChallanDoc.accountNo}</span>
            </div>

            <div className="space-y-1 text-[11px] border-b border-zinc-200 pb-2">
              <div className="flex justify-between">
                <span>Challan #:</span>
                <strong className="font-mono">{currentChallanDoc.challanNo}</strong>
              </div>
              <div className="flex justify-between">
                <span>Student:</span>
                <strong>{currentChallanDoc.studentName}</strong>
              </div>
              <div className="flex justify-between">
                <span>Roll / Class:</span>
                <span>{currentChallanDoc.rollNo} ({currentChallanDoc.className})</span>
              </div>
              <div className="flex justify-between">
                <span>Fee Month:</span>
                <strong className="text-red-600">{currentChallanDoc.month}</strong>
              </div>
              <div className="flex justify-between">
                <span>Due Date:</span>
                <strong className="font-mono text-red-600">{currentChallanDoc.dueDate}</strong>
              </div>
            </div>

            <table className="w-full text-left text-[10px] border-collapse border border-zinc-900">
              <thead>
                <tr className="bg-zinc-900 text-white font-bold uppercase">
                  <th className="p-1">Particulars</th>
                  <th className="p-1 text-right">Amount</th>
                </tr>
              </thead>
              <tbody>
                {currentChallanDoc.breakdown.map((b) => (
                  <tr key={b.title} className="border-b border-zinc-200">
                    <td className="p-1 text-zinc-800">{b.title}</td>
                    <td className="p-1 text-right font-mono font-bold">PKR {b.amount.toLocaleString()}</td>
                  </tr>
                ))}
              </tbody>
            </table>

            <div className="bg-zinc-100 p-2 border border-zinc-900 font-bold space-y-1 text-[11px]">
              <div className="flex justify-between">
                <span>Payable Within Due Date:</span>
                <span className="font-mono text-red-600">PKR {currentChallanDoc.netTotal.toLocaleString()}</span>
              </div>
              <div className="flex justify-between text-[10px] text-red-700">
                <span>After Due Date (+Fine PKR {currentChallanDoc.lateFeeFine}):</span>
                <span className="font-mono">PKR {(currentChallanDoc.netTotal + currentChallanDoc.lateFeeFine).toLocaleString()}</span>
              </div>
            </div>

            <div className="pt-4 text-center border-t border-zinc-300">
              <div className="border-t border-zinc-400 pt-1 text-[10px] text-zinc-500">
                Bank Receiving Officer Seal & Sign
              </div>
            </div>
          </div>
        ))}

      </div>
    </div>
  );
}

"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useEffect, useState } from "react";
import { Bell, Clock, ShieldCheck, CheckCircle2 } from "lucide-react";
import { examStaffApi } from "@/lib/examStaffApi";

export default function ExamNotificationsPage() {
  const [logs, setLogs] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const loadLogs = async () => {
    setLoading(true);
    setError("");
    try {
      const res = await examStaffApi.getDashboard();
      setLogs(res?.data?.recentLogs || []);
    } catch (err: any) {
      setError(err?.message || "Unable to load examination department notifications.");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    void loadLogs();
  }, []);

  return (
    <div className="space-y-6 pb-12">
      {/* Header */}
      <ModuleFrame><div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b border-zinc-200 pb-4">
        <div>
          <h1 className="text-2xl font-black text-zinc-950 flex items-center gap-2">
            <Bell className="h-6 w-6 text-school-primary" />
            Examination Department Notices & Logs
          </h1>
          <p className="text-xs text-zinc-500">
            Real-time audit log notifications for exam creation, marks entry verification, result calculations, and certificate issuances.
          </p>
        </div>
      </div></ModuleFrame>

      {error && (
        <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-4 text-xs font-semibold text-red-800">
          {error}
        </p>
      )}

      {/* Notifications / Audit Log List */}
      {loading ? (
        <p role="status" className="rounded-xl bg-white p-8 text-center text-xs text-zinc-500 border border-zinc-200">
          Loading notifications…
        </p>
      ) : logs.length > 0 ? (
        <div className="bg-white border border-zinc-200 p-5 rounded-2xl shadow-xs space-y-4 text-xs">
          <h2 className="font-bold text-zinc-950 border-b border-zinc-100 pb-2 uppercase tracking-wider text-[11px]">
            Audit Trail & Event Log History
          </h2>

          <div className="divide-y divide-zinc-100">
            {logs.map((log) => (
              <div key={log._id} className="py-3 flex items-start gap-3">
                <div className="w-8 h-8 rounded-xl bg-school-primary/10 text-school-primary flex items-center justify-center shrink-0 mt-0.5">
                  <ShieldCheck className="h-4 w-4" />
                </div>
                <div className="flex-1 space-y-0.5">
                  <p className="font-bold text-zinc-900">{log.details}</p>
                  <div className="flex items-center gap-3 text-[11px] text-zinc-500 font-medium">
                    <span>Action: <strong className="text-zinc-700">{log.action}</strong></span>
                    <span>Performed by: <strong className="text-zinc-700">{log.performedByName}</strong></span>
                  </div>
                </div>
                <span className="text-[10px] font-mono text-zinc-400 shrink-0">
                  {new Date(log.createdAt).toLocaleString()}
                </span>
              </div>
            ))}
          </div>
        </div>
      ) : (
        <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-10 text-center text-xs text-zinc-500">
          No audit notifications recorded yet.
        </div>
      )}
    </div>
  );
}

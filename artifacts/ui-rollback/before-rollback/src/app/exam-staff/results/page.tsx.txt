"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useEffect, useState } from "react";
import {
  Trophy,
  RefreshCw,
  CheckCircle2,
  Globe,
  Globe2,
  Award,
  Search,
  Zap,
} from "lucide-react";
import { examStaffApi } from "@/lib/examStaffApi";

export default function ExamResultsRankingPage() {
  const [exams, setExams] = useState<any[]>([]);
  const [classes, setClasses] = useState<any[]>([]);
  const [results, setResults] = useState<any[]>([]);

  const [selectedExamId, setSelectedExamId] = useState("");
  const [selectedClassId, setSelectedClassId] = useState("");

  const [loading, setLoading] = useState(false);
  const [processing, setProcessing] = useState(false);
  const [publishing, setPublishing] = useState(false);
  const [error, setError] = useState("");
  const [notice, setNotice] = useState<string | null>(null);

  useEffect(() => {
    Promise.all([
      examStaffApi.getSchedules(),
      examStaffApi.getClasses(),
    ])
      .then(([examRes, classRes]) => {
        const exList = examRes?.data || [];
        const clsList = classRes?.data || [];
        setExams(exList);
        setClasses(clsList);
        if (exList.length > 0) setSelectedExamId(exList[0]._id);
        if (clsList.length > 0) setSelectedClassId(clsList[0]._id);
      })
      .catch((err) => setError(err?.message || "Unable to load exam/class selection."));
  }, []);

  const loadResults = async () => {
    if (!selectedExamId || !selectedClassId) return;

    setLoading(true);
    setError("");
    try {
      const res = await examStaffApi.getResults(selectedExamId, selectedClassId);
      setResults(res?.data || []);
    } catch (err: any) {
      setError(err?.message || "Unable to load processed results.");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    void loadResults();
  }, [selectedExamId, selectedClassId]);

  const handleProcessResults = async () => {
    if (!selectedExamId || !selectedClassId) return;

    setProcessing(true);
    setError("");
    try {
      const res = await examStaffApi.processResults({
        examId: selectedExamId,
        classId: selectedClassId,
      });

      setNotice(res?.message || "Exam results processed and merit ranks calculated!");
      setTimeout(() => setNotice(null), 4000);
      await loadResults();
    } catch (err: any) {
      setError(err?.message || "Unable to process examination results.");
    } finally {
      setProcessing(false);
    }
  };

  const handleTogglePublish = async (shouldPublish: boolean) => {
    if (!selectedExamId || !selectedClassId) return;

    setPublishing(true);
    setError("");
    try {
      const res = await examStaffApi.publishResults({
        examId: selectedExamId,
        classId: selectedClassId,
        publish: shouldPublish,
      });

      setNotice(res?.message || `Exam results ${shouldPublish ? "published" : "unpublished"} successfully.`);
      setTimeout(() => setNotice(null), 4000);
      await loadResults();
    } catch (err: any) {
      setError(err?.message || "Unable to update result publication status.");
    } finally {
      setPublishing(false);
    }
  };

  const isPublishedClass = results.length > 0 && results.some((r) => r.isPublished);

  return (
    <div className="space-y-6 pb-12">
      {/* Header */}
      <ModuleFrame><div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b border-zinc-200 pb-4">
        <div>
          <h1 className="text-2xl font-black text-zinc-950 flex items-center gap-2">
            <Trophy className="h-6 w-6 text-school-primary" />
            Results & Merit Position Ranking Desk
          </h1>
          <p className="text-xs text-zinc-500">
            Process student scores across subjects, compute merit position ranks, and publish official class results.
          </p>
        </div>

        <div className="flex flex-wrap items-center gap-2 shrink-0">
          <button
            onClick={handleProcessResults}
            disabled={processing || !selectedExamId || !selectedClassId}
            className="flex items-center gap-1.5 px-4 py-2 text-xs font-bold bg-school-primary text-white rounded-xl shadow-xs hover:opacity-90 disabled:opacity-50 transition-opacity"
          >
            {processing ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Zap className="h-4 w-4" />}
            {processing ? "Ranking..." : "Process & Compute Ranks"}
          </button>

          {isPublishedClass ? (
            <button
              onClick={() => handleTogglePublish(false)}
              disabled={publishing || results.length === 0}
              className="flex items-center gap-1.5 px-4 py-2 text-xs font-bold bg-zinc-800 text-white rounded-xl shadow-xs hover:bg-zinc-900 disabled:opacity-50 transition-colors"
            >
              {publishing ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Globe2 className="h-4 w-4" />}
              Unpublish Results
            </button>
          ) : (
            <button
              onClick={() => handleTogglePublish(true)}
              disabled={publishing || results.length === 0}
              className="flex items-center gap-1.5 px-4 py-2 text-xs font-bold bg-emerald-600 text-white rounded-xl shadow-xs hover:bg-emerald-700 disabled:opacity-50 transition-colors"
            >
              {publishing ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Globe className="h-4 w-4" />}
              Publish Class Results
            </button>
          )}
        </div>
      </div></ModuleFrame>

      {error && (
        <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-4 text-xs font-semibold text-red-800">
          {error}
        </p>
      )}

      {notice && (
        <div className="bg-emerald-50 border border-emerald-200 text-emerald-800 px-4 py-3 rounded-xl text-xs font-medium flex items-center gap-2">
          <CheckCircle2 className="h-4 w-4 text-emerald-600 shrink-0" />
          <span>{notice}</span>
        </div>
      )}

      {/* Selectors Toolbar */}
      <div className="bg-white border border-zinc-200 p-4 rounded-xl grid grid-cols-1 sm:grid-cols-2 gap-4 text-xs">
        <div>
          <label className="block font-bold text-zinc-700 uppercase mb-1">Select Exam *</label>
          <select
            value={selectedExamId}
            onChange={(e) => setSelectedExamId(e.target.value)}
            className="w-full border border-zinc-300 rounded-lg p-2.5 bg-zinc-50 font-semibold focus:outline-none"
          >
            <option value="">Select Exam</option>
            {exams.map((ex) => (
              <option key={ex._id} value={ex._id}>{ex.title}</option>
            ))}
          </select>
        </div>

        <div>
          <label className="block font-bold text-zinc-700 uppercase mb-1">Select Class *</label>
          <select
            value={selectedClassId}
            onChange={(e) => setSelectedClassId(e.target.value)}
            className="w-full border border-zinc-300 rounded-lg p-2.5 bg-zinc-50 font-semibold focus:outline-none"
          >
            <option value="">Select Class</option>
            {classes.map((c) => (
              <option key={c._id} value={c._id}>{c.name}</option>
            ))}
          </select>
        </div>
      </div>

      {/* Results & Merit Position Leaderboard */}
      {loading ? (
        <p role="status" className="rounded-xl bg-white p-8 text-center text-xs text-zinc-500 border border-zinc-200">
          Loading exam results & merit positions…
        </p>
      ) : results.length > 0 ? (
        <div className="space-y-4">
          {/* Top 3 Positions Showcase */}
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
            {results.slice(0, 3).map((res, idx) => (
              <div
                key={res._id}
                className={`p-4 rounded-2xl border ${
                  idx === 0 ? "bg-amber-500/10 border-amber-300 text-amber-950" :
                  idx === 1 ? "bg-zinc-200/50 border-zinc-300 text-zinc-900" :
                  "bg-orange-500/10 border-orange-300 text-orange-950"
                } space-y-1.5`}
              >
                <div className="flex items-center justify-between">
                  <span className="text-[10px] font-black uppercase tracking-wider">
                    {idx === 0 ? "1st Position — Gold" : idx === 1 ? "2nd Position — Silver" : "3rd Position — Bronze"}
                  </span>
                  <Award className={`h-5 w-5 ${idx === 0 ? "text-amber-600" : idx === 1 ? "text-zinc-500" : "text-orange-600"}`} />
                </div>
                <h3 className="font-black text-sm truncate">{res.studentName}</h3>
                <p className="text-xs font-mono">Roll: {res.rollNumber} ({res.className})</p>
                <div className="flex justify-between items-center text-xs pt-1 border-t border-black/10 font-bold">
                  <span>Marks: {res.totalObtainedMarks} / {res.totalMaxMarks}</span>
                  <span className="text-sm font-black">{res.percentage==null?"Pending":res.percentage+"%"} ({res.overallGrade})</span>
                </div>
              </div>
            ))}
          </div>

          {/* Full Merit Leaderboard Table */}
          <div className="overflow-x-auto rounded-2xl border border-zinc-200 bg-white shadow-xs">
            <table className="w-full min-w-[750px] text-left text-xs">
              <thead className="bg-zinc-950 text-white uppercase text-[10px] font-bold">
                <tr>
                  <th className="p-3.5 text-center">Merit Rank</th>
                  <th className="p-3.5">Roll No</th>
                  <th className="p-3.5">Student Name</th>
                  <th className="p-3.5">Admission No</th>
                  <th className="p-3.5 text-center">Obtained / Max</th>
                  <th className="p-3.5 text-center">Percentage</th>
                  <th className="p-3.5 text-center">Grade</th>
                  <th className="p-3.5 text-center">Status</th>
                  <th className="p-3.5 text-center">Publication</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-zinc-200">
                {results.map((r) => (
                  <tr key={r._id} className="hover:bg-zinc-50">
                    <td className="p-3.5 text-center font-black text-sm">
                      {r.rank === 1 ? (
                        <span className="bg-amber-100 text-amber-900 px-2 py-0.5 rounded-full font-black">#1</span>
                      ) : r.rank === 2 ? (
                        <span className="bg-zinc-200 text-zinc-800 px-2 py-0.5 rounded-full font-bold">#2</span>
                      ) : r.rank === 3 ? (
                        <span className="bg-orange-100 text-orange-900 px-2 py-0.5 rounded-full font-bold">#3</span>
                      ) : (
                        <span className="text-zinc-600 font-mono">#{r.rank}</span>
                      )}
                    </td>
                    <td className="p-3.5 font-mono font-bold text-zinc-900">{r.rollNumber}</td>
                    <td className="p-3.5 font-bold text-zinc-950">{r.studentName}</td>
                    <td className="p-3.5 font-mono text-zinc-500">{r.admissionNumber || "—"}</td>
                    <td className="p-3.5 text-center font-mono font-bold text-zinc-900">
                      {r.totalObtainedMarks} / {r.totalMaxMarks}
                    </td>
                    <td className="p-3.5 text-center font-mono font-black text-school-primary">
                      {r.percentage==null?"Pending":r.percentage+"%"}
                    </td>
                    <td className="p-3.5 text-center font-black">
                      <span className="bg-zinc-100 text-zinc-800 px-2 py-0.5 rounded">{r.overallGrade}</span>
                    </td>
                    <td className="p-3.5 text-center">
                      {r.status === "PASSED" ? (
                        <span className="bg-emerald-100 text-emerald-800 text-[10px] font-bold px-2 py-0.5 rounded">PASSED</span>
                      ) : (
                        <span className="bg-red-100 text-red-800 text-[10px] font-bold px-2 py-0.5 rounded">FAILED</span>
                      )}
                    </td>
                    <td className="p-3.5 text-center">
                      {r.isPublished ? (
                        <span className="bg-emerald-100 text-emerald-800 text-[10px] font-bold px-2 py-0.5 rounded flex items-center justify-center gap-1">
                          <Globe className="h-3 w-3" /> PUBLISHED
                        </span>
                      ) : (
                        <span className="bg-zinc-100 text-zinc-500 text-[10px] font-bold px-2 py-0.5 rounded">
                          DRAFT
                        </span>
                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      ) : (
        <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-10 text-center space-y-2">
          <Trophy className="h-8 w-8 text-zinc-400 mx-auto" />
          <h3 className="font-bold text-zinc-950 text-xs">No Results Processed Yet</h3>
          <p className="text-xs text-zinc-500">
            Select an exam and class, then click &quot;Process & Compute Ranks&quot; above to generate merit position rankings.
          </p>
        </div>
      )}
    </div>
  );
}

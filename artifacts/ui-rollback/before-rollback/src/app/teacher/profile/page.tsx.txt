"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useEffect, useState } from "react";
import { useAuth } from "@/lib/auth-context";
import { teacherApi } from "@/lib/teacherApi";

export default function TeacherProfilePage() {
  const { user } = useAuth(); const [teacher, setTeacher] = useState<any>(null); const [assignments, setAssignments] = useState<any[]>([]); const [error, setError] = useState(""); const [loading, setLoading] = useState(true);
  useEffect(() => { teacherApi.getDashboardMetrics().then((response) => { setTeacher(response.data?.teacher || null); setAssignments(response.data?.teacher?.teachingAssignments || []); }).catch((e: any) => setError(e.message || "Unable to load your profile.")).finally(() => setLoading(false)); }, []);
  const name = teacher?.name || user?.name || "Teacher";
  return <div className="mx-auto max-w-5xl space-y-6 pb-12"><ModuleFrame><header className="border-b border-zinc-200 pb-5"><p className="text-xs font-semibold uppercase tracking-wider text-school-primary">Account</p><h1 className="mt-1 text-3xl font-black">My Profile</h1><p className="mt-1 text-sm text-zinc-500">Your teacher account and current teaching assignments.</p></header></ModuleFrame>
    {error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{error}</p>}
    {loading ? <p role="status" className="rounded-2xl bg-white p-6 text-sm text-zinc-500">Loading your profile…</p> : <><section className="flex flex-wrap items-center gap-5 rounded-2xl border border-zinc-200 bg-white p-6"><span className="grid h-16 w-16 place-items-center rounded-2xl bg-school-primary text-xl font-black text-white">{name.split(/\s+/).map((part: string) => part[0]).slice(0, 2).join("").toUpperCase()}</span><div><h2 className="text-xl font-bold">{name}</h2><p className="text-sm text-zinc-500">{teacher?.email || user?.email}</p><p className="mt-1 text-xs text-zinc-500">{user?.school?.name || "School"}</p></div><span className="ml-auto rounded-full bg-emerald-50 px-3 py-1 text-xs font-semibold text-emerald-700">{user?.status || "Active"}</span></section>
    <section className="grid gap-4 sm:grid-cols-2"><Info label="Qualification" value={teacher?.qualification} /><Info label="Employee ID" value={teacher?.employeeId} /><Info label="Phone" value={teacher?.phone} /><Info label="Joining date" value={teacher?.joiningDate ? new Date(teacher.joiningDate).toLocaleDateString() : ""} /></section>
    <section className="rounded-2xl border border-zinc-200 bg-white p-5"><h2 className="font-bold">Teaching assignments</h2><p className="mt-1 text-sm text-zinc-500">Assignments configured by your school administrator.</p>{assignments.length ? <div className="mt-4 overflow-x-auto"><table className="w-full min-w-[560px] text-left text-sm"><thead className="bg-zinc-50 text-xs uppercase text-zinc-500"><tr><th className="p-3">Session</th><th className="p-3">Class</th><th className="p-3">Section</th><th className="p-3">Subject</th><th className="p-3">Status</th></tr></thead><tbody>{assignments.map((item: any) => <tr key={item._id} className="border-t"><td className="p-3">{item.academicSessionId?.name || "—"}</td><td className="p-3">{item.classId?.name || "—"}</td><td className="p-3">{item.sectionId?.name || "—"}</td><td className="p-3 font-semibold">{item.subjectId?.name || "—"}</td><td className="p-3">{item.status}</td></tr>)}</tbody></table></div> : <p className="mt-4 rounded-xl bg-zinc-50 p-4 text-sm text-zinc-500">No teaching assignments yet.</p>}</section></>}
  </div>;
}
function Info({ label, value }: { label: string; value?: string | null }) { return <div className="rounded-2xl border border-zinc-200 bg-white p-5"><p className="text-xs font-semibold uppercase tracking-wide text-zinc-500">{label}</p><p className="mt-2 font-semibold">{value || "Not provided"}</p></div>; }

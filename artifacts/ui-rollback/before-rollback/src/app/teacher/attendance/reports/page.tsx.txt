"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useCallback, useEffect, useMemo, useState } from "react";
import { teacherApi } from "@/lib/teacherApi";

const id = (value: any) => String(value?._id || value || "");
export default function TeacherAttendanceReportsPage() {
  const [assignments, setAssignments] = useState<any[]>([]); const [classId, setClassId] = useState(""); const [sectionId, setSectionId] = useState("");
  const [from, setFrom] = useState(""); const [to, setTo] = useState(""); const [rows, setRows] = useState<any[]>([]); const [loading, setLoading] = useState(true); const [error, setError] = useState("");
  useEffect(() => { teacherApi.getDashboardMetrics().then((response) => { const list = response.data?.teacher?.teachingAssignments || []; setAssignments(list); if (list[0]) { setClassId(id(list[0].classId)); setSectionId(id(list[0].sectionId)); } }).catch((e: any) => setError(e.message || "Unable to load your assignments.")); }, []);
  const classSections = useMemo(() => [...new Map(assignments.map((item) => { const classValue = item.classId; const sectionValue = item.sectionId; const key = `${id(classValue)}:${id(sectionValue)}`; return [key, { key, classId: id(classValue), sectionId: id(sectionValue), label: `${classValue?.name || "Class"} · ${sectionValue?.name || "Section"}` }]; }))].map((pair: any) => pair[1]), [assignments]);
  const load = useCallback(async () => { if (!classId || !sectionId) { setRows([]); setLoading(false); return; } setLoading(true); setError(""); try { const response = await teacherApi.getAttendanceReport({ from: from || undefined, to: to || undefined, classId, sectionId }); setRows(response.data || []); } catch (e: any) { setError(e.message || "Unable to load attendance history."); } finally { setLoading(false); } }, [classId, sectionId, from, to]);
  useEffect(() => { void load(); }, [load]);
  const summary = [...rows.reduce((map: Map<string, any>, row) => { const key = id(row.studentId); const current = map.get(key) || { studentId: key, name: row.studentName, rollNumber: row.rollNumber, present: 0, absent: 0, late: 0, leave: 0, total: 0 }; current.total += 1; if (row.status === "PRESENT") current.present += 1; if (row.status === "ABSENT") current.absent += 1; if (row.status === "LATE") current.late += 1; if (row.status === "LEAVE") current.leave += 1; map.set(key, current); return map; }, new Map()).values()].map((student) => ({ ...student, percentage: student.total ? Math.round(((student.present + student.late) / student.total) * 1000) / 10 : 0 }));
  return <div className="mx-auto max-w-6xl space-y-6 pb-12"><ModuleFrame><header className="border-b border-zinc-200 pb-5"><p className="text-xs font-semibold uppercase tracking-wider text-school-primary">Attendance</p><h1 className="mt-1 text-3xl font-black">Attendance Reports</h1><p className="mt-1 text-sm text-zinc-500">Student summaries are calculated from attendance registers you submitted.</p></header></ModuleFrame>
    {error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{error}</p>}
    <section className="grid gap-3 rounded-2xl border border-zinc-200 bg-white p-4 sm:grid-cols-2 lg:grid-cols-4"><label className="grid gap-1 text-xs font-semibold">Assigned class and section<select value={classSections.find((item: any) => item.classId === classId && item.sectionId === sectionId)?.key || ""} onChange={(e) => { const selected: any = classSections.find((item: any) => item.key === e.target.value); setClassId(selected?.classId || ""); setSectionId(selected?.sectionId || ""); }} className="rounded-lg border px-3 py-2"><option value="">Select class section</option>{classSections.map((item: any) => <option key={item.key} value={item.key}>{item.label}</option>)}</select></label><label className="grid gap-1 text-xs font-semibold">From<input type="date" value={from} onChange={(e) => setFrom(e.target.value)} className="rounded-lg border px-3 py-2" /></label><label className="grid gap-1 text-xs font-semibold">To<input type="date" value={to} onChange={(e) => setTo(e.target.value)} className="rounded-lg border px-3 py-2" /></label><div className="self-end rounded-lg bg-zinc-50 px-3 py-2 text-sm">{rows.length} attendance entries</div></section>
    {loading ? <p role="status" className="rounded-xl bg-white p-6 text-sm text-zinc-500">Loading attendance history…</p> : summary.length ? <div className="overflow-x-auto rounded-2xl border border-zinc-200 bg-white"><table className="w-full min-w-[650px] text-left text-sm"><thead className="bg-zinc-50 text-xs uppercase text-zinc-500"><tr><th className="p-3">Roll</th><th className="p-3">Student</th><th className="p-3">Present</th><th className="p-3">Absent</th><th className="p-3">Late / Leave</th><th className="p-3">Attendance</th></tr></thead><tbody>{summary.map((student) => <tr key={student.studentId} className="border-t"><td className="p-3 font-mono">{student.rollNumber}</td><td className="p-3 font-semibold">{student.name}</td><td className="p-3">{student.present}</td><td className="p-3">{student.absent}</td><td className="p-3">{student.late} / {student.leave}</td><td className="p-3"><span className="font-semibold">{student.percentage}%</span><span className="ml-2 text-xs text-zinc-500">({student.total} days)</span></td></tr>)}</tbody></table></div> : <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-10 text-center"><h2 className="font-bold">No attendance history found</h2><p className="mt-1 text-sm text-zinc-500">Choose one of your assigned sections and a date range.</p></div>}
  </div>;
}

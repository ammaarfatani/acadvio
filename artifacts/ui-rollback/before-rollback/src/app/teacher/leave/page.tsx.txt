"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { FormEvent, useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import { teacherApi } from "@/lib/teacherApi";

export default function TeacherLeavePage() {
  const [leaves, setLeaves] = useState<any[]>([]); const [loading, setLoading] = useState(true); const [saving, setSaving] = useState(false); const [error, setError] = useState(""); const [notice, setNotice] = useState("");
  const load = useCallback(async () => { setLoading(true); setError(""); try { const response = await teacherApi.getLeaves(); setLeaves(response.data || []); } catch (e: any) { setError(e.message || "Unable to load your leave requests."); } finally { setLoading(false); } }, []);
  useEffect(() => { void load(); }, [load]);
  async function submit(event: FormEvent<HTMLFormElement>) { event.preventDefault(); const formElement = event.currentTarget; const form = new FormData(formElement); setSaving(true); setError(""); setNotice(""); try { const response = await teacherApi.applyLeave({ leaveType: form.get("leaveType"), startDate: form.get("startDate"), endDate: form.get("endDate"), reason: form.get("reason") }); setNotice(response.message || "Leave request submitted."); formElement.reset(); await load(); } catch (e: any) { setError(e.message || "Unable to submit your leave request."); } finally { setSaving(false); } }
  return <div className="mx-auto max-w-6xl space-y-6 pb-12"><ModuleFrame><header className="flex flex-wrap items-end justify-between gap-4 border-b border-zinc-200 pb-5"><div><p className="text-xs font-semibold uppercase tracking-wider text-school-primary">Time off</p><h1 className="mt-1 text-3xl font-black">My Leave Requests</h1><p className="mt-1 text-sm text-zinc-500">Submit a request and follow its review status.</p></div></header></ModuleFrame>
    {error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{error}</p>}{notice && <p role="status" className="rounded-xl border border-emerald-200 bg-emerald-50 p-3 text-sm text-emerald-800">{notice}</p>}
    <form onSubmit={submit} className="grid gap-3 rounded-2xl border border-zinc-200 bg-white p-5 sm:grid-cols-2 lg:grid-cols-5"><label className="grid gap-1 text-xs font-semibold">Leave type<select name="leaveType" className="rounded-lg border px-3 py-2"><option value="Casual">Casual</option><option value="Sick">Sick</option><option value="Earned">Earned</option><option value="Maternity">Maternity</option><option value="Unpaid">Unpaid</option></select></label><label className="grid gap-1 text-xs font-semibold">Start date<input name="startDate" type="date" required className="rounded-lg border px-3 py-2" /></label><label className="grid gap-1 text-xs font-semibold">End date<input name="endDate" type="date" required className="rounded-lg border px-3 py-2" /></label><label className="grid gap-1 text-xs font-semibold lg:col-span-1">Reason<input name="reason" required minLength={3} className="rounded-lg border px-3 py-2" /></label><button disabled={saving} className="inline-flex items-center justify-center gap-2 self-end rounded-lg bg-school-primary px-4 py-2 text-sm font-semibold text-white disabled:opacity-50"><Plus size={16} />{saving ? "Submitting…" : "Submit request"}</button></form>
    {loading ? <p role="status" className="rounded-xl bg-white p-6 text-sm text-zinc-500">Loading your leave requests…</p> : leaves.length ? <div className="overflow-x-auto rounded-2xl border border-zinc-200 bg-white"><table className="w-full min-w-[700px] text-left text-sm"><thead className="bg-zinc-50 text-xs uppercase text-zinc-500"><tr><th className="p-3">Type</th><th className="p-3">Dates</th><th className="p-3">Days</th><th className="p-3">Reason</th><th className="p-3">Status</th><th className="p-3">Admin comment</th></tr></thead><tbody>{leaves.map((leave) => <tr key={leave._id} className="border-t"><td className="p-3 font-semibold">{leave.leaveType}</td><td className="p-3">{new Date(leave.startDate).toLocaleDateString()} – {new Date(leave.endDate).toLocaleDateString()}</td><td className="p-3">{leave.totalDays}</td><td className="p-3">{leave.reason}</td><td className="p-3"><span className={`rounded-full px-2.5 py-1 text-xs font-semibold ${leave.status === "APPROVED" ? "bg-emerald-100 text-emerald-800" : leave.status === "REJECTED" ? "bg-red-100 text-red-800" : "bg-amber-100 text-amber-800"}`}>{leave.status}</span></td><td className="p-3 text-zinc-600">{leave.reviewComment || "—"}</td></tr>)}</tbody></table></div> : <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-10 text-center"><h2 className="font-bold">No leave requests yet</h2><p className="mt-1 text-sm text-zinc-500">Submit a request above to start.</p></div>}
  </div>;
}

"use client";
import ModuleFrame from "@/components/ModuleFrame";


import { useEffect, useState } from "react";
import Link from "next/link";
import { teacherApi } from "@/lib/teacherApi";
import { useAuth } from "@/lib/auth-context";
import { BookOpen, Layers, GraduationCap, Users, Calendar, AlertCircle, ChevronRight, FileSpreadsheet } from "lucide-react";

export default function TeacherSubjectsPage() {
  const { user } = useAuth();
  const [classes, setClasses] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    teacherApi
      .getTeacherClasses()
      .then((res) => {
        setClasses(res.data?.classes || res.data || []);
      })
      .catch((e: any) => setError(e.message || "Unable to load your assigned subjects."))
      .finally(() => setLoading(false));
  }, []);

  const base = user?.school?.slug ? `/${user.school.slug}/teacher` : "/teacher";

  // Flatten teaching assignments into subject cards
  const subjectList: any[] = [];
  classes.forEach((cls) => {
    if (cls.subjects && cls.subjects.length > 0) {
      cls.subjects.forEach((subj: any) => {
        subjectList.push({
          subjectId: subj._id,
          subjectName: subj.name,
          subjectCode: subj.code,
          className: cls.className,
          sectionName: cls.sectionName,
          classCode: cls.classCode,
          academicSessionName: cls.academicSessionName,
          isClassTeacher: cls.isClassTeacher,
          studentCount: cls.studentCount,
          teacherName: user?.name || "Assigned Teacher",
        });
      });
    } else {
      // If a class is assigned without explicit subjects (e.g. class teacher duty only)
      subjectList.push({
        subjectId: `general-${cls.classId}-${cls.sectionId}`,
        subjectName: "General Class Assignment",
        subjectCode: cls.classCode || "ALL",
        className: cls.className,
        sectionName: cls.sectionName,
        classCode: cls.classCode,
        academicSessionName: cls.academicSessionName,
        isClassTeacher: cls.isClassTeacher,
        studentCount: cls.studentCount,
        teacherName: user?.name || "Assigned Teacher",
      });
    }
  });

  return (
    <div className="mx-auto max-w-7xl space-y-6 pb-12">
      <header className="rounded-2xl border border-zinc-200 bg-white p-6 shadow-sm">
        <ModuleFrame><div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
          <div>
            <div className="inline-flex items-center gap-2 rounded-full bg-red-50 px-3 py-1 text-xs font-bold text-red-600 border border-red-100">
              <BookOpen className="h-3.5 w-3.5" />
              <span>Teaching Responsibilities</span>
            </div>
            <h1 className="mt-2 text-2xl sm:text-3xl font-black text-zinc-900">My Subjects & Classes</h1>
            <p className="mt-1 text-sm text-zinc-500">
              Overview of all subjects and sections assigned to {user?.name || "you"} for active teaching.
            </p>
          </div>
          <Link
            href={`${base}/my-classes`}
            className="inline-flex items-center gap-2 rounded-xl bg-zinc-900 px-4 py-2.5 text-sm font-semibold text-white hover:bg-zinc-800 transition-colors shrink-0"
          >
            <Layers className="h-4 w-4 text-zinc-300" />
            View Class Sections
          </Link>
        </div></ModuleFrame>
      </header>

      {error && (
        <div className="rounded-2xl border border-red-200 bg-red-50 p-4 text-sm text-red-800 flex items-start gap-3">
          <AlertCircle className="h-5 w-5 text-red-600 shrink-0 mt-0.5" />
          <p>{error}</p>
        </div>
      )}

      {loading ? (
        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
          {[1, 2, 3, 4, 5, 6].map((i) => (
            <div key={i} className="h-48 rounded-2xl bg-zinc-100 animate-pulse" />
          ))}
        </div>
      ) : subjectList.length > 0 ? (
        <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
          {subjectList.map((item, idx) => (
            <article
              key={`${item.subjectId}-${item.sectionName}-${idx}`}
              className="group relative flex flex-col justify-between rounded-2xl border border-zinc-200 bg-white p-6 shadow-sm hover:shadow-md hover:border-zinc-300 transition-all"
            >
              <div>
                <div className="flex items-start justify-between gap-3">
                  <div>
                    <span className="inline-flex items-center gap-1.5 rounded-md bg-zinc-100 px-2.5 py-1 text-xs font-mono font-bold text-zinc-700">
                      {item.subjectCode || "SUBJ"}
                    </span>
                  </div>
                  {item.isClassTeacher && (
                    <span className="inline-flex items-center gap-1 rounded-full bg-amber-100 px-2.5 py-0.5 text-xs font-bold text-amber-900">
                      <GraduationCap className="h-3.5 w-3.5" />
                      Class Teacher
                    </span>
                  )}
                </div>

                <h2 className="mt-4 text-xl font-bold text-zinc-900 group-hover:text-red-600 transition-colors">
                  {item.subjectName}
                </h2>

                <div className="mt-4 space-y-2 text-xs text-zinc-600">
                  <div className="flex items-center justify-between border-b border-zinc-100 pb-2">
                    <span className="text-zinc-400 font-medium">Assigned Class:</span>
                    <span className="font-bold text-zinc-900">{item.className}</span>
                  </div>

                  <div className="flex items-center justify-between border-b border-zinc-100 pb-2">
                    <span className="text-zinc-400 font-medium">Exact Section:</span>
                    <span className="font-bold text-zinc-900">Section {item.sectionName}</span>
                  </div>

                  <div className="flex items-center justify-between border-b border-zinc-100 pb-2">
                    <span className="text-zinc-400 font-medium">Academic Session:</span>
                    <span className="font-semibold text-zinc-800">{item.academicSessionName || "Current Session"}</span>
                  </div>

                  <div className="flex items-center justify-between border-b border-zinc-100 pb-2">
                    <span className="text-zinc-400 font-medium">Enrolled Students:</span>
                    <span className="font-semibold text-zinc-800 flex items-center gap-1">
                      <Users className="h-3.5 w-3.5 text-zinc-400" />
                      {item.studentCount} active students
                    </span>
                  </div>

                  <div className="flex items-center justify-between pt-1">
                    <span className="text-zinc-400 font-medium">Assigned Teacher:</span>
                    <span className="font-semibold text-zinc-900">{item.teacherName}</span>
                  </div>
                </div>
              </div>

              <div className="mt-6 pt-4 border-t border-zinc-100 flex items-center justify-between gap-2">
                <Link
                  href={`${base}/students?classId=${item.className}&sectionId=${item.sectionName}`}
                  className="inline-flex items-center gap-1 text-xs font-bold text-zinc-700 hover:text-zinc-900"
                >
                  View Students
                  <ChevronRight className="h-3.5 w-3.5" />
                </Link>

                <Link
                  href={`${base}/marks`}
                  className="inline-flex items-center gap-1.5 rounded-lg bg-red-50 px-3 py-1.5 text-xs font-semibold text-red-600 hover:bg-red-100 transition-colors"
                >
                  <FileSpreadsheet className="h-3.5 w-3.5" />
                  Enter Marks
                </Link>
              </div>
            </article>
          ))}
        </div>
      ) : (
        <div className="rounded-2xl border border-dashed border-zinc-300 bg-white p-12 text-center">
          <BookOpen className="mx-auto h-12 w-12 text-zinc-300" />
          <h2 className="mt-3 text-lg font-bold text-zinc-900">No assigned subjects found</h2>
          <p className="mt-1 text-sm text-zinc-500 max-w-md mx-auto">
            Your school administrator has not assigned active subject teaching duties to your teacher account yet.
          </p>
        </div>
      )}
    </div>
  );
}

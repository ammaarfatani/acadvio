"use client";
import {useState} from "react";
import {apiFetch} from "@/lib/api";
export default function ChangePassword({onClose}:{onClose:()=>void}){
 const [form,setForm]=useState({currentPassword:"",newPassword:"",confirmPassword:""}),[busy,setBusy]=useState(false),[error,setError]=useState(""),[done,setDone]=useState(false);
 async function submit(e:React.FormEvent){e.preventDefault();setBusy(true);setError("");try{await apiFetch("/auth/change-password",{method:"POST",body:JSON.stringify(form)});localStorage.removeItem("acadivo_access_token");sessionStorage.removeItem("acadivo:branch");setDone(true);}catch(e:any){setError(e.message);}finally{setBusy(false);}}
 return <div className="fixed inset-0 z-[100] grid place-items-center bg-black/30 p-4"><section role="dialog" aria-modal="true" aria-labelledby="password-heading" className="w-full max-w-md rounded-2xl border bg-white p-6 shadow-lg"><h2 id="password-heading" className="text-xl font-bold">Change password</h2>{done?<><p role="status" className="my-4 text-sm">Password changed successfully. Your previous sessions have been revoked.</p><button onClick={()=>window.location.assign("/login")} className="rounded-lg bg-red-600 px-4 py-2 text-white">Sign in again</button></>:<form onSubmit={submit} className="mt-4 space-y-4">{error&&<p role="alert" className="text-sm text-red-700">{error}</p>}{[["currentPassword","Current password"],["newPassword","New password"],["confirmPassword","Confirm new password"]].map(([key,label])=><label className="block text-sm" key={key}>{label}<input required type="password" autoComplete={key==="currentPassword"?"current-password":"new-password"} minLength={key==="currentPassword"?1:10} maxLength={72} value={form[key as keyof typeof form]} onChange={e=>setForm({...form,[key]:e.target.value})} className="mt-1 block w-full rounded-lg border p-2"/></label>)}<p className="text-xs text-zinc-500">Use 10-72 characters. You will need to sign in again.</p><div className="flex gap-3"><button disabled={busy} className="rounded-lg bg-red-600 px-4 py-2 text-sm text-white">{busy?"Updating...":"Change password"}</button><button type="button" onClick={onClose} disabled={busy} className="rounded-lg border px-4 py-2 text-sm">Cancel</button></div></form>}</section></div>;
}

"use client";
import {PageHeading} from "@/components/ProductUI";


import {useCallback,useEffect,useState,useRef} from "react";
import {apiFetch} from "@/lib/api";
import {useAuth} from "@/lib/auth-context";
import {DEFAULT_POLICY} from "@/lib/workforce-math";
import DeviceManager from "./BiometricDevices";
import {Text,Table,PayrollTable,Payslip,field,card,button,time,fmt,currentMonth} from "./WorkforceUI";
export {Payslip} from "./WorkforceUI";
export function AttendanceSetup({onComplete,mandatory=false}:any){const {refreshUser}=useAuth();const [settings,setSettings]=useState<any>(null),[mode,setMode]=useState(""),[policy,setPolicy]=useState<any>({...DEFAULT_POLICY}),[error,setError]=useState(""),[busy,setBusy]=useState(false),[saved,setSaved]=useState(false);useEffect(()=>{let live=true;apiFetch("/workforce/settings").then(r=>{if(live){setSettings(r.data);setMode(r.data.mode||"");setPolicy(r.data.policy);}}).catch(e=>setError(e.message));return()=>{live=false;};},[]);
async function save(complete:boolean){if(settings?.mode&&settings.mode!==mode&&!window.confirm("Changing attendance mode affects future attendance records. Existing attendance history will remain unchanged."))return;setBusy(true);setError("");try{const r=await apiFetch("/workforce/settings",{method:"PUT",body:JSON.stringify({mode,policy,confirmModeChange:true,complete})});setSettings(r.data);setSaved(true);await refreshUser();if(complete)onComplete?.();}catch(e:any){setError(e.message);}finally{setBusy(false);}}
return <section className={card}><h2 className="text-xl font-semibold">How does your school manage staff attendance?</h2><p className="mt-2 text-sm text-zinc-500">Choose your school&apos;s method. Existing attendance history is preserved.</p><div className="my-5 grid gap-3 sm:grid-cols-2">{[["MANUAL","Manual Attendance","Staff and teachers mark their attendance through Acadivo."],["BIOMETRIC","Biometric Attendance","Attendance is automatically synchronized from your school's fingerprint/face attendance device."]].map(([id,title,description])=><button key={id} onClick={()=>{setMode(id);setSaved(false);}} aria-pressed={mode===id} className={"rounded-xl border p-4 text-left "+(mode===id?"border-brand-600 bg-red-50":"border-zinc-200")}><b>{title}</b><p className="mt-2 text-xs leading-relaxed text-zinc-600">{description}</p></button>)}</div>{mode&&<><h3 className="mb-3 font-medium">Attendance policy</h3><div className="grid gap-4 sm:grid-cols-3">{[["startTime","Work start","time"],["endTime","Work end","time"],["workingMinutes","Daily working minutes","number"],["graceMinutes","Grace minutes","number"],["timezone","School timezone","text"]].map(([k,l,t])=><Text key={k} label={l} help={{startTime:"Expected start of the staff working day.",endTime:"Expected end of the staff working day.",workingMinutes:"Paid work minutes per day, excluding unpaid breaks.",graceMinutes:"Minutes allowed after start time before attendance becomes late.",timezone:"Time zone used for attendance and payroll dates."}[k]} type={t} value={policy[k]} onChange={(v:string)=>setPolicy({...policy,[k]:t==="number"?Number(v):v})}/>)}{[["lateMethod","Late calculation",["FROM_START","AFTER_GRACE"]],["teacherStart","Teacher start rule",["SHIFT","FIRST_CLASS"]],["rounding","Money rounding",["NEAREST","UP","DOWN"]]].map(([key,label,options]:any)=><label className="text-xs" key={key}>{label}<select className={field} value={policy[key]} onChange={e=>setPolicy({...policy,[key]:e.target.value})}>{options.map((x:string)=><option key={x}>{{FROM_START:"Count late minutes from shift start",AFTER_GRACE:"Count only minutes beyond grace",SHIFT:"Use configured staff shift",FIRST_CLASS:"Use teacher’s first scheduled lesson",NEAREST:"Round to nearest rupee",UP:"Round up to next rupee",DOWN:"Round down to whole rupee"}[x]||x}</option>)}</select></label>)}</div><div className="mt-4 flex flex-wrap gap-4">{["Sunday","Monday","Tuesday","Wednesday","Thursday","Friday","Saturday"].map((day,i)=><label key={day} className="text-xs"><input type="checkbox" checked={policy.weeklyOff.includes(i)} onChange={e=>setPolicy({...policy,weeklyOff:e.target.checked?[...policy.weeklyOff,i]:policy.weeklyOff.filter((v:number)=>v!==i)})}/> {day} off</label>)}</div><div className="my-4 flex flex-wrap gap-4">{[["lateEnabled","Late deduction"],["absenceDeduction","Absence deduction"],["paidLeave","Approved paid leave stays paid"],["unpaidLeaveDeduction","Unpaid leave deduction"]].map(([k,l])=><label key={k} className="text-xs"><input type="checkbox" checked={policy[k]} onChange={e=>setPolicy({...policy,[k]:e.target.checked})}/> {l}</label>)}</div><p className="mb-4 text-xs text-zinc-500">Once the grace period ends, choose whether late minutes start at the shift start or after the grace period. Salary deductions follow these rules.</p>{mode==="BIOMETRIC"&&<p className="mb-4 text-sm">Let&apos;s connect your biometric attendance device. Save the mode, configure its supported integration and map employees.</p>}<button disabled={busy} className={button} onClick={()=>void save(mode==="MANUAL")}>{busy?"Saving...":mode==="MANUAL"?"Enable manual attendance & continue":"Save biometric mode"}</button>{mode==="BIOMETRIC"&&(saved||settings?.mode==="BIOMETRIC")&&<><DeviceManager/><button className={button+" mt-5"} disabled={busy} onClick={()=>void save(true)}>Complete biometric setup</button></>}</>}{error&&<p role="alert" className="mt-4 text-sm text-red-700">{error}</p>}{mandatory&&!mode&&<p className="mt-3 text-xs text-zinc-500">Select an attendance method to continue.</p>}</section>;}
export function StaffSelfWorkspace({payrollOnly=false}:any){const {user}=useAuth();const [month,setMonth]=useState(currentMonth),[data,setData]=useState<any>(null),[error,setError]=useState(""),[busy,setBusy]=useState(false),[slip,setSlip]=useState<any>(null),[leave,setLeave]=useState<any>({leaveType:"Casual",startDate:"",endDate:"",reason:""});const generation=useRef(0);const load=useCallback(async()=>{const current=++generation.current;try{const r=await apiFetch("/workforce/self?month="+month);if(current===generation.current)setData(r.data);}catch(e:any){if(current===generation.current)setError(e.message);}},[month]);useEffect(()=>{setData(null);void load();},[load,user?.branchId]);async function act(action:string){setBusy(true);try{await apiFetch("/workforce/self/check-"+action,{method:"POST",body:"{}"});await load();}catch(e:any){setError(e.message);}finally{setBusy(false);}}const today=data?.records.find((r:any)=>r.date===data.today);
return <div className="space-y-5"><PageHeading eyebrow="My workspace" title={payrollOnly?"Salary & payroll":"My attendance"} description={payrollOnly?"Your salary calculations and recorded payment history.":"Your check-in records and leave requests."}/>{error&&<p role="alert" className="text-sm text-red-700">{error}</p>}<div className="max-w-xs"><Text label="Month" type="month" value={month} onChange={setMonth}/></div>{!data?<p>Loading your records...</p>:<>{!payrollOnly&&<><section className={card}><p className="text-xs text-zinc-500">Attendance Mode: {data.mode||"Not configured"}</p><h2 className="mt-3 font-semibold">Today&apos;s attendance · {data.today}</h2><div className="my-4 grid gap-3 sm:grid-cols-4">{[["Status",today?.status||"Not checked in"],["Check-in",time(today?.checkedInAt)],["Check-out",time(today?.checkedOutAt)],["Working time",today?Math.floor(today.workingMinutes/60)+"h "+today.workingMinutes%60+"m":"—"],["Late minutes",today?.lateMinutes||0]].map(([k,v])=><p key={String(k)} className="text-sm"><span className="block text-xs text-zinc-500">{k}</span>{v}</p>)}</div>{data.mode==="MANUAL"&&data.setupCompletedAt?<div className="flex gap-3"><button className={button} disabled={busy||!!today?.checkedInAt} onClick={()=>void act("in")}>Check In</button><button className={button} disabled={busy||!today?.checkedInAt||!!today?.checkedOutAt} onClick={()=>void act("out")}>Check Out</button></div>:<p className="text-sm text-zinc-500">{data.mode==="BIOMETRIC"?"Attendance synchronizes from the school's biometric integration.":"School Admin must complete attendance setup."}</p>}</section><Table headers={["Date","Status","Source","Check-in","Check-out","Working minutes","Late minutes"]} rows={data.records.map((r:any)=>[r.date,r.status,r.source,time(r.checkedInAt),time(r.checkedOutAt),r.workingMinutes,r.lateMinutes])}/><form className={card+" grid gap-3 sm:grid-cols-2"} onSubmit={async e=>{e.preventDefault();setBusy(true);try{await apiFetch("/workforce/leave",{method:"POST",body:JSON.stringify(leave)});setLeave({...leave,reason:""});await load();}catch(e:any){setError(e.message);}finally{setBusy(false);}}}><h2 className="font-semibold sm:col-span-2">Request leave</h2><select className={field} value={leave.leaveType} onChange={e=>setLeave({...leave,leaveType:e.target.value})}>{["Casual","Sick","Earned","Maternity","Unpaid"].map(x=><option key={x}>{x}</option>)}</select><Text label="Reason" value={leave.reason} onChange={(reason:string)=>setLeave({...leave,reason})}/><Text label="From" type="date" value={leave.startDate} onChange={(startDate:string)=>setLeave({...leave,startDate})}/><Text label="To" type="date" value={leave.endDate} onChange={(endDate:string)=>setLeave({...leave,endDate})}/><button className={button} disabled={busy}>Submit leave request</button></form><section className={card}><h2 className="mb-4 font-semibold">My leave requests</h2><Table headers={["Leave type","Dates","Status","Review comment"]} rows={(data.leaveRequests||[]).map((r:any)=>[r.leaveType,String(r.startDate).slice(0,10)+" – "+String(r.endDate).slice(0,10),r.status,r.reviewComment||"Awaiting review"])}/></section></>}{user?.features?.includes("FINANCE")&&<section className={card}><h2 className="font-semibold">Salary & payment history</h2><p className="my-3 text-sm text-zinc-500">{data.employment?.salaryHistory?.length?"Latest base salary: PKR "+fmt(data.employment.salaryHistory.at(-1).amount):"Salary has not been configured."}</p><PayrollTable rows={data.payroll} onView={setSlip}/></section>}</>}{slip&&<Payslip record={slip} onClose={()=>setSlip(null)}/>}</div>;}

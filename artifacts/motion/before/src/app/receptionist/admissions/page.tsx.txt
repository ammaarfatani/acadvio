"use client";



import { FormEvent, useCallback, useEffect, useMemo, useState } from "react";
import { Download, Plus, Printer, UserPlus } from "lucide-react";
import { receptionistApi } from "@/lib/receptionistApi";

const statuses = ["NEW", "CONTACTED", "FOLLOW_UP", "APPLIED", "APPROVED", "ADMITTED", "REJECTED", "CANCELLED"];
const emptyInquiry = { studentName: "", guardianName: "", guardianPhone: "", guardianEmail: "", previousSchool: "", previousClass: "", applyingClassId: "", academicSessionId: "", admissionSource: "", followUpDate: "", notes: "" };

export default function AdmissionsPage() {
  const [items, setItems] = useState<any[]>([]); const [classes, setClasses] = useState<any[]>([]); const [sessions, setSessions] = useState<any[]>([]);
  const [sections, setSections] = useState<any[]>([]); const [filters, setFilters] = useState({ search: "", status: "ALL", classId: "", from: "", to: "" });
  const [page, setPage] = useState(1); const [pages, setPages] = useState(1); const [total, setTotal] = useState(0);
  const [form, setForm] = useState(emptyInquiry); const [admitting, setAdmitting] = useState<any>(null);
  const [enrollment, setEnrollment] = useState({ dob: "", gender: "Male", sectionId: "", address: "", phone: "", fatherCnic: "", fatherOccupation: "", motherName: "", motherPhone: "", guardianEmail: "", guardianRelation: "Guardian", emergencyName: "", emergencyPhone: "", emergencyRelation: "Guardian" });
  const [loading, setLoading] = useState(true); const [saving, setSaving] = useState(false); const [error, setError] = useState(""); const [notice, setNotice] = useState("");
  const load = useCallback(async () => {
    setLoading(true); setError("");
    try { const response = await receptionistApi.getAdmissions({ ...filters, page, limit: 25 }); const data = response.data; setItems(data?.items || []); setTotal(data?.total || 0); setPages(data?.pages || 1); }
    catch (e: any) { setError(e.message || "Unable to load the admission register."); setItems([]); }
    finally { setLoading(false); }
  }, [filters, page]);
  useEffect(() => { void load(); }, [load]);
  useEffect(() => { Promise.all([receptionistApi.getClasses(), receptionistApi.getAcademicSessions()]).then(([classResponse, sessionResponse]) => { setClasses(classResponse.data || []); setSessions(sessionResponse.data || []); setForm((current) => ({ ...current, academicSessionId: sessionResponse.data?.find((item: any) => item.isCurrent)?._id || "" })); }).catch((e: any) => setError(e.message || "Unable to load school classes and sessions.")); }, []);

  const classOptions = useMemo(() => classes.map((item) => ({ id: String(item._id), name: item.name })), [classes]);
  const create = async (event: FormEvent<HTMLFormElement>) => { event.preventDefault(); setSaving(true); setError(""); setNotice(""); try { await receptionistApi.createAdmission({ ...form, followUpDate: form.followUpDate || undefined }); setForm({ ...emptyInquiry, academicSessionId: sessions.find((item) => item.isCurrent)?._id || "" }); setNotice("Admission inquiry saved."); setPage(1); await load(); } catch (e: any) { setError(e.message || "Unable to save the admission inquiry."); } finally { setSaving(false); } };
  const update = async (id: string, data: any) => { setSaving(true); setError(""); setNotice(""); try { await receptionistApi.updateAdmission(id, data); setNotice("Admission record updated."); await load(); } catch (e: any) { setError(e.message || "Unable to update admission record."); } finally { setSaving(false); } };
  const openAdmit = async (item: any) => { setAdmitting(item); setEnrollment((v) => ({ ...v, sectionId: "", guardianEmail: item.guardianEmail || "", emergencyName: item.guardianName, emergencyPhone: item.guardianPhone })); try { const response = await receptionistApi.getSections(String(item.applyingClassId?._id || item.applyingClassId)); setSections(response.data || []); } catch (e: any) { setError(e.message || "Unable to load class sections."); } };
  const admit = async (event: FormEvent<HTMLFormElement>) => { event.preventDefault(); if (!admitting) return; setSaving(true); setError(""); setNotice(""); try { await receptionistApi.admitStudent(admitting._id, enrollment); setAdmitting(null); setNotice("Student enrolled and linked to the admission register."); await load(); } catch (e: any) { setError(e.message || "Unable to enroll this student."); } finally { setSaving(false); } };
  const exportCsv = () => { const header = ["Application No", "Student", "Guardian", "Contact", "Class", "Session", "Inquiry Date", "Status", "Admission No"]; const rows = items.map((item) => [item.applicationNumber, item.studentName, item.guardianName, item.guardianPhone, item.applyingClassName, item.academicSessionName, new Date(item.inquiryDate).toLocaleDateString(), item.status, item.admissionNumber || ""]); const csv = [header, ...rows].map((row) => row.map((value) => `"${String(value ?? "").replaceAll('"', '""')}"`).join(",")).join("\r\n"); const url = URL.createObjectURL(new Blob([csv], { type: "text/csv;charset=utf-8" })); const anchor = document.createElement("a"); anchor.href = url; anchor.download = "admission-register.csv"; anchor.click(); URL.revokeObjectURL(url); };

  return <div className="space-y-6 pb-12">
    <header className="flex flex-wrap items-end justify-between gap-3 border-b border-zinc-200 pb-4"><div><p className="text-xs font-semibold uppercase tracking-wider text-school-primary">Reception</p><h1 className="mt-1 text-3xl font-black">Admission Register</h1><p className="mt-1 text-sm text-zinc-500">Real inquiries, application decisions, and linked student enrollment.</p></div><div className="flex gap-2"><button onClick={exportCsv} disabled={!items.length} className="inline-flex items-center gap-2 rounded-lg border bg-white px-3 py-2 text-sm font-semibold disabled:opacity-50"><Download size={16}/>Export CSV</button><button onClick={() => window.print()} className="inline-flex items-center gap-2 rounded-lg border bg-white px-3 py-2 text-sm font-semibold"><Printer size={16}/>Print</button></div></header>
    {error && <p role="alert" className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-800">{error}</p>}{notice && <p role="status" className="rounded-xl border border-emerald-200 bg-emerald-50 p-3 text-sm text-emerald-800">{notice}</p>}
    <details className="rounded-2xl border border-zinc-200 bg-white p-5" open><summary className="cursor-pointer list-none font-bold"><span className="inline-flex items-center gap-2"><Plus size={18}/>Record an inquiry</span></summary><form onSubmit={create} className="mt-4 grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
      <label className="grid gap-1 text-xs font-semibold">Student name *<input required minLength={2} value={form.studentName} onChange={(e) => setForm({ ...form, studentName: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Guardian name *<input required value={form.guardianName} onChange={(e) => setForm({ ...form, guardianName: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Guardian phone *<input required minLength={7} value={form.guardianPhone} onChange={(e) => setForm({ ...form, guardianPhone: e.target.value })} className="rounded-lg border px-3 py-2"/></label>
      <label className="grid gap-1 text-xs font-semibold">Guardian email<input type="email" value={form.guardianEmail} onChange={(e) => setForm({ ...form, guardianEmail: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Applying class *<select required value={form.applyingClassId} onChange={(e) => setForm({ ...form, applyingClassId: e.target.value })} className="rounded-lg border px-3 py-2"><option value="">Select class</option>{classOptions.map((item) => <option key={item.id} value={item.id}>{item.name}</option>)}</select></label><label className="grid gap-1 text-xs font-semibold">Academic session *<select required value={form.academicSessionId} onChange={(e) => setForm({ ...form, academicSessionId: e.target.value })} className="rounded-lg border px-3 py-2"><option value="">Select session</option>{sessions.map((item) => <option key={item._id} value={item._id}>{item.name}{item.isCurrent ? " (Current)" : ""}</option>)}</select></label>
      <label className="grid gap-1 text-xs font-semibold">Previous school<input value={form.previousSchool} onChange={(e) => setForm({ ...form, previousSchool: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Previous class<input value={form.previousClass} onChange={(e) => setForm({ ...form, previousClass: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Source<input value={form.admissionSource} onChange={(e) => setForm({ ...form, admissionSource: e.target.value })} placeholder="Walk-in, referral, web" className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Follow-up date<input type="date" value={form.followUpDate} onChange={(e) => setForm({ ...form, followUpDate: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold sm:col-span-2">Notes<textarea value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} className="rounded-lg border px-3 py-2" rows={2}/></label><button disabled={saving || !classes.length || !sessions.length} className="self-end rounded-lg bg-school-primary px-4 py-2.5 text-sm font-bold text-white disabled:opacity-50">{saving ? "Saving…" : "Save inquiry"}</button>
    </form></details>
    <section className="grid gap-3 rounded-2xl border border-zinc-200 bg-white p-4 sm:grid-cols-2 lg:grid-cols-5"><label className="grid gap-1 text-xs font-semibold">Search<input value={filters.search} onChange={(e) => { setPage(1); setFilters({ ...filters, search: e.target.value }); }} placeholder="Student, guardian, application" className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Status<select value={filters.status} onChange={(e) => { setPage(1); setFilters({ ...filters, status: e.target.value }); }} className="rounded-lg border px-3 py-2"><option>ALL</option>{statuses.map((value) => <option key={value}>{value}</option>)}</select></label><label className="grid gap-1 text-xs font-semibold">Class<select value={filters.classId} onChange={(e) => { setPage(1); setFilters({ ...filters, classId: e.target.value }); }} className="rounded-lg border px-3 py-2"><option value="">All classes</option>{classOptions.map((item) => <option key={item.id} value={item.id}>{item.name}</option>)}</select></label><label className="grid gap-1 text-xs font-semibold">From<input type="date" value={filters.from} onChange={(e) => { setPage(1); setFilters({ ...filters, from: e.target.value }); }} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">To<input type="date" value={filters.to} onChange={(e) => { setPage(1); setFilters({ ...filters, to: e.target.value }); }} className="rounded-lg border px-3 py-2"/></label></section>
    <section className="overflow-x-auto rounded-2xl border border-zinc-200 bg-white"><table className="w-full min-w-[1100px] text-left text-sm"><thead className="bg-zinc-50 text-xs uppercase text-zinc-500"><tr><th className="p-3">Application</th><th className="p-3">Inquiry date</th><th className="p-3">Student / Guardian</th><th className="p-3">Class / Session</th><th className="p-3">Documents</th><th className="p-3">Status</th><th className="p-3">Enrollment</th></tr></thead><tbody>{loading ? <tr><td colSpan={7} className="p-8 text-center text-zinc-500">Loading admission records…</td></tr> : !items.length ? <tr><td colSpan={7} className="p-10 text-center text-zinc-500">No admission inquiries found.</td></tr> : items.map((item) => <tr key={item._id} className="border-t"><td className="p-3 font-mono font-semibold">{item.applicationNumber}<div className="text-xs text-zinc-500">{item.admissionNumber || "Not admitted"}</div></td><td className="p-3">{new Date(item.inquiryDate).toLocaleDateString()}</td><td className="p-3"><strong>{item.studentName}</strong><div className="text-xs text-zinc-500">{item.guardianName} · {item.guardianPhone}</div></td><td className="p-3">{item.applyingClassName}<div className="text-xs text-zinc-500">{item.academicSessionName}</div></td><td className="p-3"><select disabled={["ADMITTED", "REJECTED", "CANCELLED"].includes(item.status) || saving} value={item.documentsStatus} onChange={(e) => void update(item._id, { documentsStatus: e.target.value })} className="rounded border px-2 py-1 text-xs">{["PENDING", "PARTIAL", "COMPLETE"].map((value) => <option key={value}>{value}</option>)}</select></td><td className="p-3"><select disabled={["ADMITTED", "REJECTED", "CANCELLED"].includes(item.status) || saving} value={item.status} onChange={(e) => void update(item._id, { status: e.target.value })} className="rounded border px-2 py-1 text-xs">{statuses.map((value) => <option key={value}>{value}</option>)}</select></td><td className="p-3">{item.status === "APPROVED" ? <button onClick={() => void openAdmit(item)} className="inline-flex items-center gap-1 rounded-lg bg-school-primary px-3 py-2 text-xs font-bold text-white"><UserPlus size={14}/>Enroll</button> : item.status === "ADMITTED" && item.studentId ? <span className="text-xs font-semibold text-emerald-700">Linked to student</span> : <span className="text-xs text-zinc-400">Approve to enroll</span>}</td></tr>)}</tbody></table></section>
    <footer className="flex items-center justify-between text-sm text-zinc-500"><span>{total} admission records</span><div className="flex gap-2"><button disabled={page <= 1 || loading} onClick={() => setPage(page - 1)} className="rounded border bg-white px-3 py-1.5 disabled:opacity-50">Previous</button><span className="px-2 py-1.5">Page {page} of {Math.max(1, pages)}</span><button disabled={page >= pages || loading} onClick={() => setPage(page + 1)} className="rounded border bg-white px-3 py-1.5 disabled:opacity-50">Next</button></div></footer>
    {admitting && <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4"><form onSubmit={admit} className="max-h-[90vh] w-full max-w-2xl space-y-4 overflow-y-auto rounded-2xl bg-white p-6"><div><h2 className="text-xl font-black">Enroll {admitting.studentName}</h2><p className="text-sm text-zinc-500">Enrollment creates the student record in this school and links it to this application.</p></div><div className="grid gap-3 sm:grid-cols-2"><label className="grid gap-1 text-xs font-semibold">Date of birth *<input required type="date" value={enrollment.dob} onChange={(e) => setEnrollment({ ...enrollment, dob: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Gender *<select value={enrollment.gender} onChange={(e) => setEnrollment({ ...enrollment, gender: e.target.value })} className="rounded-lg border px-3 py-2"><option>Male</option><option>Female</option><option>Other</option></select></label><label className="grid gap-1 text-xs font-semibold">Section *<select required value={enrollment.sectionId} onChange={(e) => setEnrollment({ ...enrollment, sectionId: e.target.value })} className="rounded-lg border px-3 py-2"><option value="">Select section</option>{sections.map((section) => <option key={section._id} value={section._id}>{section.name}</option>)}</select></label><label className="grid gap-1 text-xs font-semibold">Student phone<input value={enrollment.phone} onChange={(e) => setEnrollment({ ...enrollment, phone: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold sm:col-span-2">Address *<textarea required minLength={5} value={enrollment.address} onChange={(e) => setEnrollment({ ...enrollment, address: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Mother name<input value={enrollment.motherName} onChange={(e) => setEnrollment({ ...enrollment, motherName: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Mother phone<input value={enrollment.motherPhone} onChange={(e) => setEnrollment({ ...enrollment, motherPhone: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Emergency contact name<input value={enrollment.emergencyName} onChange={(e) => setEnrollment({ ...enrollment, emergencyName: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Emergency contact phone<input value={enrollment.emergencyPhone} onChange={(e) => setEnrollment({ ...enrollment, emergencyPhone: e.target.value })} className="rounded-lg border px-3 py-2"/></label><label className="grid gap-1 text-xs font-semibold">Emergency relation<input value={enrollment.emergencyRelation} onChange={(e) => setEnrollment({ ...enrollment, emergencyRelation: e.target.value })} className="rounded-lg border px-3 py-2"/></label></div><div className="flex justify-end gap-2 border-t pt-4"><button type="button" onClick={() => setAdmitting(null)} className="rounded-lg border px-4 py-2 text-sm">Cancel</button><button disabled={saving || !sections.length} className="rounded-lg bg-school-primary px-4 py-2 text-sm font-bold text-white disabled:opacity-50">{saving ? "Enrolling…" : "Create enrollment"}</button></div></form></div>}
  </div>;
}

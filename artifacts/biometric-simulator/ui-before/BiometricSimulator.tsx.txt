"use client";
import {useCallback,useEffect,useState} from "react";
import {apiFetch} from "@/lib/api";
import {useAuth} from "@/lib/auth-context";
import {Text,Table,field,button,fmt,currentMonth,time} from "./WorkforceUI";
const localInput=(at:Date)=>new Date(at.getTime()-at.getTimezoneOffset()*60000).toISOString().slice(0,16);
export default function BiometricSimulator({device,employees,onChanged}:{device:any;employees:any[];onChanged:()=>Promise<void>}){
 const {user}=useAuth();
 const [open,setOpen]=useState(false),[runId,setRunId]=useState<string>(device.lastDemoRunId||""),[employeeId,setEmployee]=useState(""),[month,setMonth]=useState(currentMonth()),[date,setDate]=useState(new Date().toISOString().slice(0,10)),[timestamp,setTimestamp]=useState(""),[eventId,setEventId]=useState(""),[lastBatch,setLastBatch]=useState<any>(null),[report,setReport]=useState<any>(null),[result,setResult]=useState<any>(null),[busy,setBusy]=useState(false),[loading,setLoading]=useState(false),[error,setError]=useState("");
 const authorized=user?.userType==="SCHOOL_ADMIN"&&["biometric.manage","staff-attendance.manage"].every(p=>user.permissions?.includes(p));
 const mapped=employees.filter(e=>device.mappings?.some((m:any)=>String(m.employeeId)===String(e._id)));
 useEffect(()=>{setRunId(previous=>previous||crypto.randomUUID());setTimestamp(localInput(new Date()));setEventId(crypto.randomUUID());},[device._id]);
 const mappedIds=mapped.map(e=>String(e._id)).join(",");
 useEffect(()=>{setEmployee(previous=>mappedIds.split(",").includes(previous)?previous:mappedIds.split(",")[0]||"");setReport(null);setLastBatch(null);},[device._id,mappedIds]);
 const loadReport=useCallback(async()=>{if(!employeeId||!runId||!/^([a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12})$/i.test(runId))return;const r=await apiFetch("/workforce/devices/"+device._id+"/demo/report?"+new URLSearchParams({runId,employeeId,month}));return r.data;},[device._id,employeeId,runId,month]);
 useEffect(()=>{if(!open||!employeeId||!runId)return;let live=true;setLoading(true);setReport(null);loadReport().then(data=>{if(live)setReport(data);}).catch(e=>{if(live)setError(e.message);}).finally(()=>{if(live)setLoading(false);});return()=>{live=false;};},[open,loadReport,employeeId,runId]);
 if(!authorized)return null;
 const mapping=device.mappings?.find((m:any)=>String(m.employeeId)===employeeId);
 const day=report?.rows.find((r:any)=>r.date===date);
 async function toggle(){setBusy(true);setError("");try{const r=await apiFetch("/workforce/devices/"+device._id+"/demo",{method:"PUT",body:JSON.stringify({enabled:!device.demoEnabled})});setResult(r.data);await onChanged();}catch(e:any){setError(e.message);}finally{setBusy(false);}}
 async function submit(batch?:any){if(!mapping&&!batch)return;
 if(!batch){const at=Date.parse(timestamp);if(!Number.isFinite(at)||at>Date.now()+300000||at<Date.parse("2000-01-01")){setError("Choose a valid past/current punch time. For check-out testing use a completed date; more than five minutes in the future is not allowed.");return;}if(!/^([a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12})$/i.test(runId)){setError("Enter a valid saved run UUID, or choose New isolated run.");return;}}
 setBusy(true);setError("");try{
   const body=batch||{runId,events:[{eventId,biometricUserId:mapping.biometricUserId,timestamp:new Date(timestamp).toISOString()}]};
   const r=await apiFetch("/workforce/devices/"+device._id+"/demo/events",{method:"POST",body:JSON.stringify(body)});setResult(r.data);setLastBatch(body);setReport(await loadReport());await onChanged();
 }catch(e:any){setError(e instanceof RangeError?"Choose a valid punch date and time.":e.message);}finally{setBusy(false);}}
 function chooseTime(kind:"in"|"late"|"grace"|"out"){
   if(!day)return;const p=day.policySnapshot;const at=new Date(kind==="out"?day.scheduledEnd:day.scheduledStart);
   if(kind==="late")at.setMinutes(at.getMinutes()+p.graceMinutes+1);
   if(kind==="grace")at.setMinutes(at.getMinutes()+p.graceMinutes);
   setTimestamp(localInput(at));setEventId(crypto.randomUUID());
 }
 return <section className="mt-4 border-t border-dashed pt-4"><button type="button" aria-expanded={open} onClick={()=>setOpen(!open)} className="text-sm font-semibold text-brand-600">DEMO / Testing Mode · {device.demoEnabled?"Enabled":"Disabled"}</button>{open&&<div className="mt-3 space-y-4"><p className="rounded border border-amber-200 bg-amber-50 p-3 text-xs leading-6 text-amber-900">No physical device required. Test punches use the real mapping and attendance engine. DEMO snapshots are excluded from real attendance and payroll. Testing never proves hardware connectivity.</p><p className="text-xs">Real integration status: {device.connectionStatus} · Testing: {device.demoEnabled?"Ready":"Disabled"} · Last test: {device.lastDemoAt?new Date(device.lastDemoAt).toLocaleString():"Not tested"}</p><button disabled={busy||!user?.branchId} type="button" className="rounded border px-3 py-2 text-xs" onClick={()=>void toggle()}>{device.demoEnabled?"Disable testing":"Enable DEMO testing"}</button>{!user?.branchId&&<p className="text-xs text-amber-800">Select a specific branch before testing.</p>}{!mapped.length&&<p className="text-sm text-zinc-500">Map an existing staff member or teacher to this device first.</p>}
 <div className="grid gap-3 sm:grid-cols-2"><label className="text-xs">Mapped employee<select className={field} value={employeeId} onChange={e=>{setEmployee(e.target.value);setLastBatch(null);}}><option value="">Choose mapped employee</option>{mapped.map(e=><option key={e._id} value={e._id}>{e.name} · {e.userType}</option>)}</select></label><Text label="Report month" type="month" value={month} onChange={(v:string)=>{setMonth(v);setDate(v+"-01");}}/><Text label="Scenario date" type="date" value={date} onChange={setDate}/><Text label="Punch time (your browser timezone)" type="datetime-local" value={timestamp} onChange={setTimestamp}/><Text label="Event ID (keep unchanged to retry a duplicate)" value={eventId} onChange={setEventId}/><Text label="DEMO run ID (restore a saved test run)" value={runId} onChange={(v:string)=>{setRunId(v);setLastBatch(null);}}/></div>
 <p className="text-xs text-zinc-500">Biometric ID: {mapping?.biometricUserId||"Not mapped"}. First punch becomes check-in; the latest distinct punch becomes check-out, matching the actual integration.</p><div className="flex flex-wrap gap-2">{([["in","Normal check-in"],["grace","At grace boundary"],["late","Late arrival"],["out","Check-out"]] as const).map(([kind,label])=><button type="button" key={kind} disabled={!day||busy} className="rounded border px-3 py-2 text-xs" onClick={()=>chooseTime(kind)}>{label}</button>)}<button type="button" disabled={busy} className="rounded border px-3 py-2 text-xs" onClick={()=>{setRunId(crypto.randomUUID());setEventId(crypto.randomUUID());setLastBatch(null);setResult(null);setReport(null);}}>New isolated run</button></div>
 <div className="flex flex-wrap gap-3"><button type="button" className={button} disabled={busy||loading||!device.demoEnabled||!device.enabled||!mapping||!eventId||!timestamp} onClick={()=>void submit()}>{busy?"Processing test…":"Send DEMO punch"}</button><button type="button" className="rounded border px-3 py-2 text-xs" disabled={busy||!lastBatch||!device.demoEnabled} onClick={()=>void submit(lastBatch)}>Replay last batch (duplicate test)</button></div>
 {error&&<p role="alert" className="text-sm text-red-700">{error}</p>}{result&&<div role="status" className="rounded border bg-zinc-50 p-3 text-xs">{result.message||`DEMO: ${result.processed} processed · ${result.duplicates} duplicates · ${result.errors?.length||0} need review`}{result.errors?.map((e:any)=><p className="mt-2 text-red-700" key={e.eventId}>{e.message}</p>)}</div>}
 {loading&&<p role="status" className="text-xs text-zinc-500">Loading DEMO report…</p>}{report&&<><h4 className="text-sm font-semibold">DEMO attendance report · {report.employee.name}</h4><p className="text-xs text-zinc-500">Run {report.runId} · School policy timezone {day?.policySnapshot?.timezone||"configured timezone"} · Not genuine biometric attendance</p><Table headers={["Date","Day","DEMO status","Check-in","Check-out","Late min","Work min"]} rows={report.rows.map((r:any)=>[r.date,r.dayType,r.status,time(r.checkedInAt),time(r.checkedOutAt),r.lateMinutes,r.workingMinutes])}/><p className="text-xs">{report.previewMessage}</p>{report.deductionPreview&&<div className="rounded border border-amber-200 bg-amber-50 p-3 text-xs"><b>DEMO deduction preview — never saved to payroll</b><p className="mt-2">Late PKR {fmt(report.deductionPreview.lateDeduction)} · Absence PKR {fmt(report.deductionPreview.absenceDeduction)} · Weekly-off days excluded from payable days ({report.deductionPreview.payableDays}).</p></div>}</>}
 </div>}</section>;
}
